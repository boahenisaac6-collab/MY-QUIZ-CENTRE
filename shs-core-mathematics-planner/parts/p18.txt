ts.org/officeDocument/2006/relationships', wp: 'http://schemas.openxmlformats.org/drawingml/2006/wordprocessingDrawing',
    a: 'http://schemas.openxmlformats.org/drawingml/2006/main', pic: 'http://schemas.openxmlformats.org/drawingml/2006/picture'
  };
  const ACC = '7A1F6E';
  const esc = s => String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');

  function build(d, state) {
    // state: {lists: next num id, nums: [], images: [], docPr}
    const rr = (s, o, sz) => `<w:r><w:rPr>${o.b ? '<w:b/><w:bCs/>' : ''}${o.i ? '<w:i/><w:iCs/>' : ''}${o.color ? `<w:color w:val="${o.color}"/>` : ''}<w:sz w:val="${sz}"/><w:szCs w:val="${sz}"/></w:rPr><w:t xml:space="preserve">${esc(s)}</w:t></w:r>`;
    function runs(text, o = {}) {
      const sz = Math.round((o.size || 10) * 2);
      return inline(text).map(x => x.t === 'math' ? `<m:oMath>${TeX.omml(x.tex, sz)}</m:oMath>` : rr(x.s, { b: x.b || o.b, i: o.i, color: o.color }, sz)).join('');
    }
    function para(inner, o = {}) {
      const ppr = [
        '', // keep-with-next is not used: some editors then refuse to split the long lesson rows
        o.num ? `<w:numPr><w:ilvl w:val="0"/><w:numId w:val="${o.num}"/></w:numPr>` : '',
        o.rule ? '<w:pBdr><w:bottom w:val="single" w:sz="6" w:space="2" w:color="C99BC2"/></w:pBdr>' : o.bar ? '<w:pBdr><w:left w:val="single" w:sz="24" w:space="4" w:color="7A1F6E"/></w:pBdr>' : '',
        o.shade ? `<w:shd w:val="clear" w:color="auto" w:fill="${o.shade}"/>` : '',
        o.tab ? `<w:tabs><w:tab w:val="right" w:pos="${o.tab}"/></w:tabs>` : '',
        `<w:spacing w:before="${Math.round((o.before || 0) * 20)}" w:after="${Math.round((o.after ?? 2) * 20)}" w:line="252" w:lineRule="auto"/>`,
        o.num ? '' : `<w:ind w:left="${o.bar ? 80 : 0}" w:firstLine="0"/>`,
        `<w:jc w:val="${o.align || 'left'}"/>`
      ].join('');
      return `<w:p><w:pPr>${ppr}</w:pPr>${inner}</w:p>`;
    }
    function newList(kind) {
      const id = state.next++;
      state.nums.push({ id, kind });
      return id;
    }
    function blocks(list, width) { return (list || []).filter(Boolean).map(b => block(b, width)).join(''); }
    function block(b, width) {
      const o = { size: b.size || 10, color: b.color === 'accent' ? ACC : null, i: b.italic };
      switch (b.k) {
        case 'p': return para(runs(b.text, o), { keep: b.keep, before: b.before || 0, after: b.after ?? 2, shade: b.note ? 'FDF1DE' : b.adapt ? 'E4EEFA' : null });
        case 'h': return para(runs(b.text, { size: b.lvl === 1 ? 10.8 : 10, b: true, color: b.lvl === 1 ? ACC : null }), { keep: true, before: b.before ?? (b.lvl === 1 ? 5 : 3), after: b.rule ? 3 : 2, rule: b.rule });
        case 'bar': {
          const tab = Math.max(3000, (width || 12000) - 200);
          return para(runs(`**${b.text}**`, { size: 10.2 }) + (b.right ? '<w:r><w:tab/></w:r>' + rr(b.right, { color: '595959' }, 17) : ''), { shade: 'F7E8F4', bar: true, tab, before: b.before ?? 6, after: 3 });
        }
        case 'ul': case 'ol': {
          const num = newList((b.k === 'ul' ? 'bullet' : 'decimal') + (b.sub ? '-sub' : ''));
          return b.items.map(it => para(runs(it, o), { num, after: 1 })).join('');
        }
        case 'math': return `<w:p><w:pPr><w:spacing w:before="40" w:after="40"/><w:jc w:val="center"/></w:pPr><m:oMathPara><m:oMath>${TeX.omml(b.tex, 20)}</m:oMath></m:oMathPara></w:p>`;
        case 'fig': {
          const f = FIGS[b.id]; if (!f) return '';
          // a narrow diagram floats to the right of the text that follows it, carrying its caption (as in the PDF)
          const float = f.doc && (b.cm || 8) * 567 <= 0.46 * width;
          const useDoc = !!f.doc;            // the picture carries its own caption, so the frame holds both
          const cx = Math.round((b.cm || 8) * 0.92 * 360000);
          const cy = Math.round(cx * (useDoc ? f.dh / f.dw : f.h / f.w));
          const key = String(b.id).replace(/[^\w-]/g, '_') + (useDoc ? '-doc' : '');
          let img = state.images.find(x => x.key === key);
          if (!img) { img = { key, id: b.id, doc: useDoc, rid: 'rIdImg' + (state.images.length + 1), file: `media/${key}.png` }; state.images.push(img); }
          const n = ++state.docPr;
          const frame = '<a:ln w="9525"><a:solidFill><a:srgbClr val="B8B0BC"/></a:solidFill></a:ln>';
          const graphic = `<a:graphic><a:graphicData uri="http://schemas.openxmlformats.org/drawingml/2006/picture"><pic:pic><pic:nvPicPr><pic:cNvPr id="${n}" name="${key}.png"/><pic:cNvPicPr/></pic:nvPicPr><pic:blipFill><a:blip r:embed="${img.rid}"/><a:stretch><a:fillRect/></a:stretch></pic:blipFill><pic:spPr><a:xfrm><a:off x="0" y="0"/><a:ext cx="${cx}" cy="${cy}"/></a:xfrm><a:prstGeom prst="rect"><a:avLst/></a:prstGeom>${frame}</pic:spPr></pic:pic></a:graphicData></a:graphic>`;
          const descr = esc(TeX.plain(b.caption.replace(/\$/g, '')));
          if (float) {
            const anchor = `<w:r><w:drawing><wp:anchor distT="0" distB="45720" distL="182880" distR="0" simplePos="0" relativeHeight="${251658240 + n}" behindDoc="0" locked="0" layoutInCell="1" allowOverlap="0"><wp:simplePos x="0" y="0"/><wp:positionH relativeFrom="column"><wp:align>right</wp:align></wp:positionH><wp:positionV relativeFrom="paragraph"><wp:posOffset>0</wp:posOffset></wp:positionV><wp:extent cx="${cx}" cy="${cy}"/><wp:effectExtent l="0" t="0" r="0" b="0"/><wp:wrapSquare wrapText="left"/><wp:docPr id="${n}" name="Figure ${n}" descr="${descr}"/><wp:cNvGraphicFramePr><a:graphicFrameLocks noChangeAspect="1"/></wp:cNvGraphicFramePr>${graphic}</wp:anchor></w:drawing></w:r>`;
            return `<w:p><w:pPr><w:spacing w:before="0" w:after="0" w:line="20" w:lineRule="exact"/><w:ind w:left="0" w:firstLine="0"/></w:pPr>${anchor}</w:p>`;
          }
          const drawing = `<w:r><w:drawing><wp:inline distT="0" distB="0" distL="0" distR="0"><wp:extent cx="${cx}" cy="${cy}"/><wp:effectExtent l="0" t="0" r="0" b="0"/><wp:docPr id="${n}" name="Figure ${n}" descr="${descr}"/><wp:cNvGraphicFramePr><a:graphicFrameLocks noChangeAspect="1"/></wp:cNvGraphicFramePr>${graphic}</wp:inline></w:drawing></w:r>`;
          return para(drawing, { align: 'center', before: 6, after: useDoc ? 6 : 0 }) + (useDoc ? '' : para(runs(b.caption, { size: 9, i: true }), { align: 'center', after: 4 }));
        }
        case 'table': {
          const W = width - 120;
          const cw = b.cols.map(p => Math.floor(W * p / 100));
          const border = ['top', 'left', 'bottom', 'right', 'insideH', 'insideV'].map(s => `<w:${s} w:val="single" w:sz="4" w:space="0" w:color="A6A6A6"/>`).join('');
          const tc = (content, i, head) => `<w:tc><w:tcPr><w:tcW w:w="${cw[i]}" w:type="dxa"/>${head ? `<w:shd w:val="clear" w:color="auto" w:fill="${b.fill || 'F2CEED'}"/>` : ''}</w:tcPr>${content}</w:tc>`;
          const cellC = (c, sz) => Array.isArray(c) ? blocks(c, cw[0]) : para(runs(c, { size: sz }), { after: 0 });
          let x = `<w:tbl><w:tblPr><w:tblW w:w="${cw.reduce((a, b) => a + b, 0)}" w:type="dxa"/><w:tblBorders>${border}</w:tblBorders><w:tblLayout w:type="fixed"/><w:tblCellMar><w:top w:w="30" w:type="dxa"/><w:left w:w="80" w:type="dxa"/><w:bottom w:w="30" w:type="dxa"/><w:right w:w="80" w:type="dxa"/></w:tblCellMar><w:tblLook w:val="04A0" w:firstRow="1" w:lastRow="0" w:firstColumn="1" w:lastColumn="0" w:noHBand="0" w:noVBand="1"/></w:tblPr><w:tblGrid>${cw.map(w => `<w:gridCol w:w="${w}"/>`).join('')}</w:tblGrid>`;
          if (b.head) x += `<w:tr><w:trPr><w:cantSplit/><w:tblHeader/></w:trPr>${b.head.map((h, i) => tc(para(runs(h, { size: 9.5, b: true, color: ACC }), { after: 0, align: b.small ? 'center' : 'left' }) + (b.sub ? para(runs(b.sub[i], { size: 8.5, i: true }), { after: 0 }) : ''), i, true)).join('')}</w:tr>`;
          for (const row of b.rows) x += `<w:tr>${row.map((c, i) => tc(Array.isArray(c) ? blocks(c, cw[i]) : para(runs(c, { size: 9 }), { after: 0, align: b.small ? 'center' : 'left' }), i, false)).join('')}</w:tr>`;
          return x + '</w:tbl>' + para('', { after: 2 });
        }
      }
      return '';
    }
    return { runs, para, blocks };
  }

  async function make(d, templateB64) {
    const zip = await JSZip.loadAsync(templateB64, { base64: true });
    const xml = await zip.file('word/document.xml').async('string');
    const doc = new DOMParser().parseFromString(xml, 'application/xml');
    const W = NS.w;
    const kids = (el, name) => Array.from(el.childNodes).filter(n => n.nodeType === 1 && n.localName === name);
    const body = doc.getElementsByTagNameNS(W, 'body')[0];
    const tbl = kids(body, 'tbl')[0];
    let rows = kids(tbl, 'tr');
    const state = { next: 950, nums: [], images: [], docPr: 100 };
    const R = build(d, state);
    const wrap = inner => `<x xmlns:w="${NS.w}" xmlns:m="${NS.m}" xmlns:r="${NS.r}" xmlns:wp="${NS.wp}" xmlns:a="${NS.a}" xmlns:pic="${NS.pic}">${inner}</x>`;
    function fill(tc, innerXml) {
      for (const n of Array.from(tc.childNodes)) if (!(n.nodeType === 1 && n.localName === 'tcPr')) tc.removeChild(n);
      const frag = new DOMParser().parseFromString(wrap(innerXml || R.para('')), 'application/xml');
      const err = frag.getElementsByTagName('parsererror')[0];
      if (err) throw new Error('XML: ' + err.textContent.slice(0, 200));
      const list = Array.from(frag.documentElement.childNodes);
      for (const n of list) tc.appendChild(doc.importNode(n, true));
      const last = Array.from(tc.childNodes).filter(n => n.nodeType === 1).pop();
      if (!last || last.localName !== 'p') tc.appendChild(doc.importNode(new DOMParser().parseFromString(wrap(R.para('')), 'application/xml').documentElement.firstChild, true));
    }
    const cells = tr => kids(tr, 'tc');
    const FULL = 12730, CONTENT = 10070;
    // row 0: subject line
    fill(cells(rows[0])[0], R.para(R.runs('**Name of Subject:** ' + d.subject, { size: 11 }) + (d.extra.length ? R.runs('      ' + d.extra.join('   ·   '), { size: 9.5 }) : ''), { after: 0 }));
    fill(cells(rows[1])[0], R.para(R.runs('**' + d.heading + '**', { size: 12 }), { align: 'center', after: 0 }));
    const meta = cells(rows[2]);
    [[1, d.learners], [3, d.week], [5, d.duration], [7, d.form]].forEach(([i, v]) => fill(meta[i], R.para(R.runs(v, { size: 10 }), { after: 0 })));
    const content = [d.strand, d.substrand, d.standard, d.outcome, d.indicators, d.eqs, d.strategies, d.resources];
    content.forEach((c, k) => fill(cells(rows[3 + k])[1], R.blocks(c, CONTENT)));
    fill(cells(rows[12])[0], R.blocks(d.diff, FULL));
    // lessons: duplicate rows 13–15 for each lesson
    const proto = [rows[13], rows[14], rows[15]];
    let anchor = rows[15];
    const blocksets = [proto];
    for (let i = 1; i < d.lessons.length; i++) {
      const c = proto.map(r => r.cloneNode(true));
      c.forEach(r => { anchor.parentNode.insertBefore(r, anchor.nextSibling); anchor = r; });
      blocksets.push(c);
    }
    d.lessons.forEach((l, i) => {
      const [a, b, c] = blocksets[i];
      const colon = l.title.indexOf(':');
      fill(cells(a)[0], R.para(R.runs('**' + l.title.slice(0, colon) + ':**', { size: 11 }) + R.runs(' ' + l.title.slice(colon + 1).trim(), { size: 11, b: true }), { align: 'center', keep: true, after: 0 }));
      fill(cells(b)[0], R.para(R.runs(l.ref, { size: 9, i: true }), { align: 'center', keep: true, after: 0 }));
      fill(cells(c)[0], R.blocks(l.blocks, FULL));
    });
    rows = kids(tbl, 'tr');
    const ka = rows.findIndex(r => r.textContent.includes('Key Assessment') && r.textContent.length < 40);
    fill(cells(rows[ka + 1])[0], R.blocks(d.assessment, FULL));
    const rf = rows.findIndex(r => r.textContent.trim().startsWith('Reflection'));
    fill(cells(rows[rf + 1])[0], R.blocks(d.reflection, FULL));
    // relabel the lesson rows' hint text is replaced above; serialise
    let out = new XMLSerializer().serializeToString(doc);
    if (!out.startsWith('<?xml')) out = '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n' + out;
    zip.file('word/document.xml', out);
    // numbering: one bullet and one decimal abstract, a num per list (restart)
    let num = await zip.file('word/numbering.xml').async('string');
    const abs = (id, fmt, txt, font) => `<w:abstractNum w:abstractNumId="${id}"><w:multiLevelType w:val="hybridMultilevel"/><w:lvl w:ilvl="0"><w:start w:val="1"/><w:numFmt w:val="${fmt}"/><w:lvlText w:val="${txt}"/><w:lvlJc w:val="left"/><w:pPr><w:ind w:left="300" w:hanging="240"/></w:pPr>${font ? `<w:rPr><w:rFonts w:ascii="${font}" w:hAnsi="${font}" w:hint="default"/></w:rPr>` : ''}</w:lvl></w:abstractNum>`;
    const abs2 = (id, fmt, txt, left, font) => abs(id, fmt, txt, font).replace('<w:ind w:left="300" w:hanging="240"/>', `<w:ind w:left="${left}" w:hanging="260"/>`);
    const absX = abs(940, 'bullet', '•', 'Arial') + abs(941, 'decimal', '%1.', null) + abs2(942, 'bullet', '–', 620, 'Arial') + abs2(943, 'decimal', '%1.', 660, null);
    const ABS = { bullet: 940, decimal: 941, 'bullet-sub': 942, 'decimal-sub': 943 };
    const numX = state.nums.map(n => `<w:num w:numId="${n.id}"><w:abstractNumId w:val="${ABS[n.kind] || 940}"/><w:lvlOverride w:ilvl="0"><w:startOverride w:val="1"/></w:lvlOverride></w:num>`).join('');
    const firstNum = num.indexOf('<w:num ');
    num = firstNum >= 0 ? num.slice(0, firstNum) + absX + num.slice(firstNum) : num.replace('</w:numbering>', absX + '</w:numbering>');
    num = num.replace('</w:numbering>', numX + '</w:numbering>');
    zip.file('word/numbering.xml', num);
    // images
    if (state.images.length) {
      let rels = await zip.file('word/_rels/document.xml.rels').async('string');
      for (const im of state.images) {
        zip.file('word/' + im.file, im.doc ? FIGS[im.id].doc : FIGS[im.id].png, { base64: true });
        rels = rels.replace('</Relationships>', `<Relationship Id="${im.rid}" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/image" Target="${im.file}"/></Relationships>`);
      }
      zip.file('word/_rels/document.xml.rels', rels);
      let ct = await zip.file('[Content_Types].xml').async('string');
      if (!/Extension="png"/i.test(ct)) ct = ct.replace('<Default Extension="xml"', '<Default Extension="png" ContentType="image/png"/><Default Extension="xml"');
      zip.file('[Content_Types].xml', ct);
    }
    // document title in core properties
    const core = zip.file('docProps/core.xml');
    if (core) {
      let c = await core.async('string');
      c = c.replace(/<dc:title>[^<]*<\/dc:title>|<dc:title\/>/, '').replace('</cp:coreProperties>', `<dc:title>${esc('Learning Planner – ' + d.subject + ' – Year ' + (d.year || 1) + ', Week ' + d.week)}</dc:title></cp:coreProperties>`);
      zip.file('docProps/core.xml', c);
    }
    return zip.generateAsync({ type: 'blob', mimeType: 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', compression: 'DEFLATE' });
  }
  return { make };
})();

/* =============================================================== HTML for pasting into Word / WPS */
const PASTE = (() => {
  const st = {
    table: 'border-collapse:collapse;width:100%;font-family:\'Times New Roman\',serif;font-size:10pt',
    td: 'border:1px solid #000;padding:3pt 5pt;vertical-align:top',
    lab: 'border:1px solid #000;padding:3pt 5pt;vertical-align:top;background:#F2CEED;font-weight:bold',
    deep: 'border:1px solid #000;padding:3pt 5pt;vertical-align:top;background:#E59EDC;font-weight:bold',
    inner: 'border-collapse:collapse;width:100%;font-size:9pt', itd: 'border:1px solid #999;padding:2pt 4pt;vertical-align:top',
    ith: 'border:1px solid #999;padding:2pt 4pt;vertical-align:top;background:#F2CEED;color:#7A1F6E;font-weight:bold'
  };
  const runs = t => inline(t).map(r => r.t === 'math' ? TeX.html(r.tex) : r.b ? `<b>${escH(r.s)}</b>` : escH(r.s)).join('');
  const P = (inner, extra = '') => `<p style="margin:0 0 2pt;${extra}">${inner}</p>`;
  function block(b) {
    if (!b) return '';    const size = b.size ? `font-size:${b.size}pt;` : '';
    const shade = b.note ? 'background:#FDF1DE;' : b.adapt ? 'background:#E4EEFA;' : '';
    const col = b.color === 'accent' ? 'color:#7A1F6E;' : '';
    switch (b.k) {
      case 'p': return P(runs(b.text), size + shade + col);
      case 'h': return P(`<b>${runs(b.text)}</b>`, `margin-top:6pt;${b.lvl === 1 ? 'color:#7A1F6E;font-size:10.5pt;' : ''}${b.rule ? 'border-bottom:1px solid #C99BC2;' : ''}`);
      case 'bar': return P(`<b>${runs(b.text)}</b>${b.right ? ` <span style="color:#595959;font-size:8.5pt">— ${escH(b.right)}</span>` : ''}`, 'margin-top:6pt;background:#F7E8F4;border-left:3px solid #7A1F6E;padding:2pt 4pt');
      case 'ul': case 'ol': return `<${b.k} style="margin:0 0 2pt ${b.sub ? 34 : 18}pt;padding:0;${size}${b.sub ? (b.k === 'ol' ? 'list-style-type:decimal;' : 'list-style-type:circle;') : ''}">${b.items.map(i => `<li>${runs(i)}</li>`).join('')}</${b.k}>`;
      case 'fig': { const f = FIGS[b.id]; if (!f) return ''; const w = Math.round((b.cm || 8) * 37.8); return `<p style="text-align:center;margin:4pt 0 0"><img width="${w}" height="${Math.round(w * f.h / f.w)}" style="border:1px solid #B8B0BC;padding:4px" src="data:image/png;base64,${f.png}"></p>` + P(`<i>${runs(b.caption)}</i>`, 'text-align:center;font-size:9pt'); }
      case 'table': {
        const cell = c => Array.isArray(c) ? c.map(block).join('') : runs(c);
        const head = b.head ? `<tr>${b.head.map((h, i) => `<td style="${st.ith};width:${b.cols[i]}%">${runs(h)}${b.sub ? `<br><i style="font-weight:normal;color:#333">${runs(b.sub[i])}</i>` : ''}</td>`).join('')}</tr>` : '';
        return `<table style="${st.inner}">${head}${b.rows.map(r => `<tr>${r.map(c => `<td style="${st.itd}">${cell(c)}</td>`).join('')}</tr>`).join('')}</table>`;
      }
    }
    return '';
  }
  const blocks = l => (l || []).map(block).join('');
  function planner(d) {
    const row = (label, list, deep) => `<tr><td style="${deep ? st.deep : st.lab};width:20%">${label}</td><td colspan="7" style="${st.td}">${blocks(list)}</td></tr>`;
    const full = (inner, s2) => `<tr><td colspan="8" style="${s2 || st.td}">${inner}</td></tr>`;
    return `<table style="${st.table}">
      ${full(`<b>Name of Subject:</b> ${escH(d.subject)}${d.extra.length ? ' &nbsp;&nbsp; ' + d.extra.map(escH).join(' · ') : ''}`)}
      ${full(`<b>${escH(d.heading)}</b>`, st.td + ';background:#F6C5AC;text-align:center')}
      <tr><td style="${st.lab}">Number of Learners in Class</td><td style="${st.td}">${escH(d.learners)}</td><td style="${st.lab}">Week</td><td style="${st.td}">${escH(d.week)}</td><td style="${st.deep}">Duration</td><td style="${st.td}">${escH(d.duration)}</td><td style="${st.lab}">Form</td><td style="${st.td}">${escH(d.form)}</td></tr>
      ${row('Strand', d.strand)}${row('Sub-Strand', d.substrand)}${row('Content Standard', d.standard)}${row('Learning Outcome(s)', d.outcome)}
      ${row('Learning Indicator(s)', d.indicators)}${row('Essential Question(s)', d.eqs, true)}${row('Pedagogical Strategies', d.strategies)}${row('Teaching &amp; Learning Resources', d.resources)}
      ${full('<b>Key Notes on Differentiation</b>', st.lab)}${full(blocks(d.diff))}
      ${d.lessons.map(l => full(`<b>${runs(l.title)}</b>`, st.lab + ';text-align:center') + full(`<i>${runs(l.ref)}</i>`, st.lab + ';text-align:center;font-weight:normal;font-size:9pt') + full(blocks(l.blocks))).join('')}
      ${full('<b>Key Assessment</b>', st.lab)}${full(blocks(d.assessment))}
      ${full('<b>Reflection &amp; Remarks</b>', st.lab)}${full(blocks(d.reflection))}
    </table>`;
  }
  return { planner };
})();

const Smart = (() => {
  const WORDNUM = { one: 1, two: 2, three: 3, four: 4, five: 5, six: 6, seven: 7, eight: 8, nine: 9, ten: 10, eleven: 11, twelve: 12, thirteen: 13, fourteen: 14, fifteen: 15, sixteen: 16, seventeen: 17, eighteen: 18, nineteen: 19, twenty: 20, thirty: 30, forty: 40, fifty: 50, sixty: 60, seventy: 70, eighty: 80, ninety: 90, a: 1, an: 1, single: 1 };
  const NEED_WORDS = {
    visual: /(blind|visual(ly)?|low vision|poor sight|sight problem|short[- ]sighted|albin\w*|cannot see|can't see)/,
    hearing: /(deaf|hearing|hard of hearing)/,
    physical: /(wheelchair|crutch\w*|physical(ly)?|cerebral|limb|cannot walk|lame|motor)/,
    reading: /(dyslexi\w*|cannot read|can't read|poor read\w*|reading (difficult\w*|problem\w*)|struggl\w* to read)/,
    attention: /(adhd|hyperactiv\w*|attention|restless|can't sit|cannot sit|distract\w*)/,
    language: /(limited english|weak english|poor english|little english|english is (weak|poor)|second[- ]language|french[- ]speaking|language barrier)/,
    shy: /(shy|quiet|timid|withdrawn)/,
    gifted: /(gifted|talented|brilliant|very (good|bright|fast)|fast learners?|olympiad)/,
    absent: /(absent\w*|latecom\w*|come late|comes late|truan\w*|irregular)/,
    behaviour: /(disrupt\w*|noisy|troublesome|stubborn|behaviou?r (problem|issue)\w*|misbehav\w*)/
  };
  const RES_WORDS = {
    projector: /(projector|beamer)/, calculators: /(calculators?)/, phones: /(phones?|tablets?|smartphones?)/,
    internet: /(internet|wifi|wi-fi|geogebra|data bundle)/, printer: /(printer|photocop\w*|printing)/,
    manila: /(manila|cardboard)/, slates: /(slates?|mini[- ]?whiteboards?|whiteboards)/, sticky: /(sticky notes?|post-?its?)/,
    tape: /(masking tape|tape)/, maths_set: /(math(ematical)?s? sets?|geometry sets?|compass(es)?)/, material: /(learner material|textbooks?|learner'?s? book)/,
    tactile: /(tactile|braille)/
  };
  const LANGS = ['Twi', 'Fante', 'Ewe', 'Ga', 'Dangme', 'Krobo', 'Dagbani', 'Hausa', 'Nzema', 'Gonja', 'Kasem', 'Dagaare', 'Akuapem', 'Asante', 'Bono', 'Sefwi', 'Gurune', 'Likpakpaanl', 'French'];
  const MONTHS = ['jan', 'feb', 'mar', 'apr', 'may', 'jun', 'jul', 'aug', 'sep', 'oct', 'nov', 'dec'];
  const DAYS = ['monday', 'tuesday', 'wednesday', 'thursday', 'friday'];

  function words2num(t) { return t.replace(/\b(one|two|three|four|five|six|seven|eight|nine|ten|eleven|twelve|thirteen|fourteen|fifteen|sixteen|seventeen|eighteen|nineteen|twenty|thirty|forty|fifty|sixty|seventy|eighty|ninety)(?:[- ](one|two|three|four|five|six|seven|eight|nine))?\b/g, (m, a, b) => String(WORDNUM[a] + (b ? WORDNUM[b] : 0))); }
  const iso = d => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;

  /* ---------------------------------------------- offline reader */
  function readClass(text, S) {
    const raw = String(text || '');
    const t = ' ' + words2num(raw.toLowerCase()).replace(/[’']/g, "'") + ' ';
    const set = {}, found = [];
    const num = re => { const m = re.exec(t); return m ? +m[1] : null; };
    let boys = num(/(\d+)\s*(?:of them\s*)?(?:are\s*)?(?:boys|males?|male (?:students|learners))\b/);
    let girls = num(/(\d+)\s*(?:of them\s*)?(?:are\s*)?(?:girls|females?|female (?:students|learners))\b/);
    // a class size, not "two learners have low vision": skip counts followed by a need or by have/with/who
    let total = num(/class (?:of|has|with) (\d+)\b/);
    if (total == null) {
      const re = /(\d+)\s*(?:students|learners|pupils|children|kids)\b(.{0,40})/g; let m;
      while ((m = re.exec(t))) { const after = m[2]; if (+m[1] < 8) continue; if (/^\s*(have|has|with|who|are|is|use|cannot|can't|need)\b/.test(after)) continue; if (Object.values(NEED_WORDS).some(r => r.test(after.slice(0, 25)))) continue; total = +m[1]; break; }
    }
    if (/(all[- ]girls|girls'? (school|class)|single[- ]sex girls|only girls)/.test(t)) { boys = 0; if (girls == null && total != null) girls = total; }
    if (/(all[- ]boys|boys'? (school|class)|single[- ]sex boys|only boys)/.test(t)) { girls = 0; if (boys == null && total != null) boys = total; }
    if (boys != null && girls == null && total != null && total >= boys) girls = total - boys;
    if (girls != null && boys == null && total != null && total >= girls) boys = total - girls;
    if (boys == null && girls == null && total != null) {
      const b0 = +S.boys || 1, g0 = +S.girls || 1; boys = Math.round(total * b0 / (b0 + g0)); girls = total - boys;
      found.push(`${total} learners (I kept your boys–girls ratio: ${boys} boys, ${girls} girls)`);
    } else if (boys != null || girls != null) found.push(`${boys ?? S.boys} boys, ${girls ?? S.girls} girls`);
    if (boys != null) set.boys = boys; if (girls != null) set.girls = girls;

    // lesson lengths
    const lens = [];
    t.replace(/(\d+(?:\.\d+)?)\s*(?:-|\s)?\s*(hours?|hrs?|min|mins|minutes)\b/g, (m, a, u) => { const v = /^h/.test(u) ? Math.round(+a * 60) : +a; if (v >= 30 && v <= 180) lens.push(v); return m; });
    if (/\b1 hour each|an hour each|each (?:lesson )?(?:is )?(?:an|1) hour/.test(t) && lens.length === 1) lens.push(lens[0]);
    const per = [];
    t.replace(/\b(single|double|triple)\b(?:\s*period)?/g, (m, a) => { per.push({ single: 40, double: 80, triple: 120 }[a]); return m; });
    const L = lens.length ? lens : per;
    if (/\bboth\b[^.]{0,30}\b(double|single|triple)/.test(t) || /\b(two|2) (double|single|triple) periods?/.test(t)) {
      const v = { single: 40, double: 80, triple: 120 }[(t.match(/\b(double|single|triple)\b/) || [])[1]]; if (v) { set.dur1 = set.dur2 = v; found.push(`both lessons ${v} min`); }
    } else if (L.length) {
      set.dur1 = L[0]; set.dur2 = L[1] ?? L[0];
      found.push(L.length > 1 ? `Lesson 1: ${set.dur1} min, Lesson 2: ${set.dur2} min` : `${set.dur1}-minute lessons`);
    }
    // days and date
    const days = []; t.replace(/\b(monday|tuesday|wednesday|thursday|friday)s?\b/g, (m, d) => { days.push(DAYS.indexOf(d)); return m; });
    if (days.length) { set.day1 = String(days[0]); if (days[1] != null) set.day2 = String(days[1]); found.push('lesson days: ' + days.slice(0, 2).map(d => DAYS[d][0].toUpperCase() + DAYS[d].slice(1)).join(' and ')); }
    const dm = /\b(\d{1,2})(?:st|nd|rd|th)?\s+(jan|feb|mar|apr|may|jun|jul|aug|sep|sept|oct|nov|dec)\w*,?\s+(\d{4})\b/.exec(t) || /\b(jan|feb|mar|apr|may|jun|jul|aug|sep|sept|oct|nov|dec)\w*\s+(\d{1,2})(?:st|nd|rd|th)?,?\s+(\d{4})\b/.exec(t);
    if (dm) {
      const [d, mo, y] = /^\d/.test(dm[1]) ? [+dm[1], dm[2], +dm[3]] : [+dm[2], dm[1], +dm[3]];
      const date = new Date(y, MONTHS.indexOf(mo.slice(0, 3)), d);
      if (!isNaN(date)) { set.weekStart = iso(date); found.push('week starting ' + date.toDateString().slice(4)); }
    }
    // needs, with counts when a number is close before the need words
    const needs = new Set(S.needs || []), counts = { ...(S.needCounts || {}) };
    for (const [k, re] of Object.entries(NEED_WORDS)) {
      const g = new RegExp(re.source, 'g'); let m, hit = null, cnt = null;
      while ((m = g.exec(t))) {
        if (k === 'attention' && /pay attention to|attention to/.test(t.slice(Math.max(0, m.index - 5), m.index + 25))) continue;
        hit = hit || m;
        const before = t.slice(Math.max(0, m.index - 40), m.index).split(/[.;,]/).pop();
        const c = /(\d+)\s+(?:[a-z'-]+\s+){0,4}$/.exec(before);
        if (c && +c[1] < 40) { cnt = +c[1]; break; }
      }
      if (!hit) continue;
      needs.add(k);
      if (cnt) counts[k] = cnt;
      found.push(`${cnt ? cnt + ' ' : ''}${NEEDS[k].toLowerCase()}`);
    }
    if (needs.size !== (S.needs || []).length || JSON.stringify(counts) !== JSON.stringify(S.needCounts || {})) { set.needs = [...needs]; set.needCounts = counts; }
    // resources: "no X" / "without X" removes, "have X" adds
    const res = new Set(S.res || []);
    let resChanged = false;
    if (/(no resources|only (the )?(chalk|board|chalkboard|marker)|nothing but (the )?board)/.test(t)) { res.clear(); resChanged = true; found.push('board only'); }
    for (const [k, re] of Object.entries(RES_WORDS)) {
      const m = new RegExp(re.source, 'g'); let x;
      while ((x = m.exec(t))) {
        const before = t.slice(Math.max(0, x.index - 28), x.index);
        const neg = /(no|not|without|lack|don't have|do not have|never|none of|neither|nor)\s+(\w+\s+){0,3}$/.test(before) || /\b(no|not)\s+$/.test(before);
        if (neg) { if (res.has(k)) { res.delete(k); resChanged = true; } found.push(`no ${RESOURCES[k].toLowerCase()}`); }
        else { if (!res.has(k)) { res.add(k); resChanged = true; } found.push(RESOURCES[k].toLowerCase()); }
        break;
      }
    }
    if (resChanged) set.res = [...res];
    // programme and class name
    const progs = [['science', /\b(general )?science\b|\bsci\b/], ['business', /\bbusiness\b|\bbus\b/], ['agric', /\bagric\w*/], ['homeec', /\bhome ?econ\w*|\bh\.?e\b/], ['visual', /\bvisual arts?\b|\bv\.?a\b/], ['technical', /\btechnical\b|\btech\b/], ['arts', /\b(general )?arts?\b/]];
    for (const [k, re] of progs) if (re.test(t)) { set.programme = k; found.push(PROGRAMMES[k]); break; }
    const cn = /\b(?:form\s*)?(1|2|3|one|two|three)\s*(sci(?:ence)?|arts?|bus(?:iness)?|agric|h\.?e|home\s*econ\w*|v\.?a|visual\s*arts?|tech\w*)\s*([a-z]|\d)?\b/i.exec(raw);
    if (cn) { const map = { sci: 'Science', science: 'Science', art: 'Arts', arts: 'Arts', bus: 'Business', business: 'Business', agric: 'Agric', he: 'Home Economics', 'h.e': 'Home Economics', va: 'Visual Arts', 'v.a': 'Visual Arts', tech: 'Technical', technical: 'Technical' }; const key = cn[2].toLowerCase().replace(/\s+/g, ''); const nm = map[key] || (key.startsWith('home') ? 'Home Economics' : key.startsWith('visual') ? 'Visual Arts' : key[0].toUpperCase() + key.slice(1)); set.className = `${({ one: 1, two: 2, three: 3 })[cn[1].toLowerCase()] || cn[1]} ${nm}${cn[3] ? ' ' + cn[3].toUpperCase() : ''}`; found.push('class ' + set.className); }
    // year and week of the curriculum
    const ym = /\b(?:year|yr|shs|form)\s*(1|2|3)\b/.exec(t);
    const yr = ym ? +ym[1] : (set.className ? +set.className[0] : null);
    const wm = /\bweek\s*(\d{1,2})\b/.exec(t);
    if (yr || wm) {
      const cur = /^y(\d)w(\d\d)$/.exec(String(S.week)) || [null, '1', '01'];
      const Y = yr || +cur[1], Wn = wm ? +wm[1] : (yr && yr !== +cur[1] ? 1 : +cur[2]);
      if (Wn >= 1 && Wn <= (Y === 3 ? 20 : 24)) { set.week = `y${Y}w${String(Wn).padStart(2, '0')}`; found.push(`Year ${Y}, Week ${Wn}`); }
    }
    // readiness
    const avg = /(?:average|mean|scored?|score of|got)\D{0,12}(\d{1,3})\s*(?:%|percent)/.exec(t);
    if (avg && +avg[1] <= 100) { set.bandMode = 'score'; set.lastScore = +avg[1]; found.push(`last test average ${avg[1]}% (readiness estimated)`); }
    else if (/(weak|struggl\w*|below average|poor (class|performance)|low ability|slow learners)/.test(t)) { set.bandMode = 'manual'; set.ap = 55; set.p = 35; set.hp = 10; found.push('mostly approaching proficiency (55% AP)'); }
    else if (/(strong|above average|bright class|brilliant class|very good class|high ability|excellent class|clever class)/.test(t)) { set.bandMode = 'manual'; set.ap = 15; set.p = 50; set.hp = 35; found.push('a strong class (35% HP)'); }
    // seating, language, term, school
    if (/(fixed|cannot move|can't move|bolted|long) (desks|benches|seats|furniture)|(desks|benches|seats) (are )?(fixed|cannot be moved)/.test(t)) { set.seating = 'fixed'; found.push('fixed benches'); }
    else if (/(movable|moveable|can move) (desks|chairs|furniture)|(desks|chairs) can (be )?move/.test(t)) { set.seating = 'movable'; found.push('movable desks'); }
    for (const L2 of LANGS) if (new RegExp('\\b' + L2.toLowerCase() + '\\b').test(t)) { set.lang = L2 === 'Krobo' ? 'Dangme (Krobo)' : L2; found.push('familiar language: ' + set.lang); break; }
    const sm = /\b(?:semester|term)\s*(1|2|one|two)\b/.exec(t); if (sm) { set.term = sm[1] === 'one' ? '1' : sm[1] === 'two' ? '2' : sm[1]; found.push('Semester ' + set.term); }
    const sch = /\b(?:school is|i teach at|at|from)\s+([A-Z][\w.'&-]*(?:\s+[A-Z][\w.'&/-]*){0,6}\s+(?:SHS|SHTS|Senior High(?:\/Technical)?(?: School)?|Secondary School|High School|College))/.exec(raw);
    if (sch) { set.school = sch[1].trim(); found.push('school: ' + set.school); }
    if (/(no homework|don't give homework|without homework)/.test(t)) { set.homework = false; found.push('no homework'); }
    return { set, found: [...new Set(found)] };
  }

  /* ---------------------------------------------- what the planner decided, and why */
  function insights(doc, S) {
    const { ctx, lessons, W } = doc.summary;
    const M = W.meta || { label: 'Year 1, Week 1', nEx: 7, exLabel: i => `Worked Example 1.${i + 1}`, flags: [] };
    const b = ctx.band, dec = [], adv = [], q = [];
    const pct = x => Math.round(x * 100);
    // decisions
    dec.push(['groups', `${ctx.n} learners${ctx.movable ? ' with movable desks' : ' on fixed benches'} → ${ctx.G} groups of ${ctx.size}${ctx.autoGroups ? (ctx.n > 40 ? ' (groups of 6 keep the number of groups small enough to visit each one)' : ctx.n <= 24 ? ' (small class: groups of 4 give everyone a role)' : ' (groups of 5: one role each)') : ''}. Each group gets a mix of HP, P and AP learners${ctx.single || ctx.paired ? '' : ' and of girls and boys'} — see the group plan.`]);
    if (b.known) dec.push(['readiness', `${b.source === 'score' ? `Your last test average of ${b.score}% suggests` : 'Your readiness mix gives'} about ${b.ap} AP, ${b.p} P and ${b.hp} HP learners.${ctx.apHeavy ? ' Because more than 45% are AP, extra “I do – we do” modelling is added to four activities.' : ''}${ctx.hpHeavy ? ' Because many learners are HP, one HP learner per group acts as explainer.' : ''}`]);
    else dec.push(['readiness', 'Readiness is unknown, so Lesson 1 starts with a diagnostic and readiness groups are formed from the first exit ticket.']);
    if (!ctx.single) {
      const g = ctx.girlShare;
      dec.push(['gender', ctx.paired ? `${ctx.paired.who === 'girls' ? 'Girls' : 'Boys'} are only ${ctx.paired.who === 'girls' ? pct(g) : pct(1 - g)}% of the class → they are seated at least two to a group (${ctx.paired.mixed} of ${ctx.paired.groups} groups) so no one is alone, and they report first in alternate tasks.` : g < 0.4 ? `Girls are ${pct(g)}% of the class → every group has at least two girls and girls report first in alternate tasks.` : g > 0.6 ? `Boys are ${pct(1 - g)}% of the class → every group has at least two boys.` : `The class is gender-balanced (${pct(g)}% girls) → girls and boys share the leader and reporter roles equally.`]);
    } else dec.push(['gender', `A single-sex class (${ctx.single}) → roles rotate so every learner leads and reports once in the week.`]);
    lessons.forEach((x, i) => {
      const dr = x.timing.dropped;
      if (dr.length) dec.push(['time', `Lesson ${i + 1} is ${x.D} minutes → ${dr.map(it => (x.timing.carriedOut.includes(it.id) ? `${it.title.replace(/ \(.*\)$/, '')} moves to Lesson ${i + 2} (planned there)` : `${it.title.replace(/ \(.*\)$/, '')} ${it.ph === 'main' ? 'becomes homework' : 'is folded into another part'}`)).join('; ')}.`]);
      else if (x.D > 80) dec.push(['time', `Lesson ${i + 1} is ${x.D} minutes → an extension activity is added and every activity gets more time.`]);
    });
    dec.push(['resources', (M.handmade ? [
      ctx.has('projector') ? 'Projector → the diagrams are projected' : (ctx.has('manila') ? 'No projector → the diagrams go on manila charts prepared before the lesson' : 'No projector or manila → the diagrams are drawn on the board before the lesson'),
      `${ctx.has('printer') ? 'Printer → number cards are printed' : 'No printer → number cards are cut by hand'} (${ctx.G} sets, one per group)`,
      ctx.has('calculators') ? 'calculators are used only to check long division' : 'no calculators → decimals are found by long division',
      ctx.has('tape') ? 'masking tape or chalk → a human number line on the floor' : 'the number line is drawn along the board',
      ctx.has('slates') ? 'mini whiteboards for show-me answers' : 'jotters are held up for show-me answers'
    ] : [
      ctx.has('projector') ? 'Projector → the figures are projected' : (ctx.has('manila') ? 'No projector → the figures go on manila charts prepared before the lesson' : 'No projector or manila → the figures are drawn on the board before the lesson'),
      `${ctx.has('printer') ? 'Printer → task cards are printed' : 'No printer → task cards are cut by hand'} (${ctx.G} sets, one per group)`,
      ctx.has('calculators') ? 'calculators are used to check answers, not to replace the method' : 'no calculators → working is done by hand (four-figure tables where needed)',
      ctx.has('slates') ? 'mini whiteboards for show-me answers' : 'jotters are held up for show-me answers'
    ]).join('; ') + '.']);
    const needs = [...ctx.needs];
    if (needs.length) dec.push(['needs', `${needs.length} learner need${needs.length > 1 ? 's' : ''} ticked → each gets a strategy in the differentiation notes and an “Inclusion in this activity” line inside the activities where it matters${needs.some(k => ['visual', 'hearing', 'physical', 'attention'].includes(k)) ? '; the group plan places these learners (front groups for low vision and hearing, an aisle seat for physical disability)' : ''}.`]);
    if (M.handmade) dec.push(['context', `${ctx.progName} → the daily-life task uses ${({ business: 'profit, loss, discount and shop-floor', science: 'laboratory and measurement', agric: 'farm and harvest', homeec: 'kitchen and baking', visual: 'canvas, scale and design', technical: 'workshop, tolerance and frame', arts: 'voting, timeline and athletics-track', mixed: 'everyday' })[ctx.prog] || 'everyday'} examples.`]);
    if (ctx.lang) dec.push(['language', `Familiar language ${ctx.lang} → second-language learners may explain first in ${ctx.lang}, then in English.`]);
    const kaItems = doc.assessment.filter(x => x.k === 'ol').flatMap(x => x.items).filter(i => /DoK/.test(i));
    const ka = kaItems.length ? { items: kaItems } : null;
    if (ka) dec.push(['assessment', M.handmade ? `${S.keyMins} minutes for the key assessment → ${ka.items.length} of the Teacher Manual’s Week 1 key-assessment items chosen to fit the time (Section A and Section B), each with a full mark scheme and performance levels.` : `${S.keyMins} minutes for the key assessment → ${ka.items.length} items from the ${M.label} Learner Material${M.tmTasks ? ' and Teacher Manual' : ''} chosen to fit the time, each with a mark scheme and performance levels.`]);
    if (M.flags && M.flags.length) adv.push({ id: 'tm', text: `For ${M.label} the Teacher Manual wording of the ${M.flags.includes('header') ? 'content standard, learning outcome and ' : ''}indicators was not available here, so it is taken from the Learner Material outcomes. Check it against your Teacher Manual.`, label: null });
    if (ctx.lessonDate[0] || ctx.lessonDate[1]) dec.push(['dates', `Week of ${ctx.weekRange} → lesson dates are written on each lesson heading.`]);
    if (S.ai && S.ai.lessons) dec.push(['claude', `Claude’s adaptations for “${(S.ai.request || '').slice(0, 60)}” are included and marked “Adaptation”.`]);

    // suggestions with one-tap fixes
    const L1 = lessons[0], L2 = lessons[1];
    if (L1.D < L2.D && L1.timing.dropped.some(it => it.ph === 'main')) adv.push({ id: 'swap', text: `Lesson 1 has more to teach than Lesson 2 but is the shorter lesson. Swap the lengths so Lesson 1 gets ${L2.D} minutes?`, label: 'Swap lengths', apply: { dur1: L2.D, dur2: L1.D } });
    if (L2.timing.dropped.some(it => it.ph === 'main' && !it.extra) && L2.D < 80) adv.push({ id: 'short2', text: `Lesson 2 (${L2.D} min) cannot fit every activity, so some become homework. A double period (80 min) fits the whole week.`, label: 'Use 80 min', apply: { dur2: 80 } });
    if (ctx.G > 10) { const bigger = Math.min(8, Math.ceil(ctx.n / 9)); adv.push({ id: 'groups', text: `${ctx.G} groups is a lot to supervise in one lesson. Groups of ${bigger} give ${Math.round(ctx.n / bigger)} groups.`, label: `Groups of ${bigger}`, apply: { groupSize: String(bigger) } }); }
    if (ctx.need('visual') && !ctx.has('tactile')) adv.push({ id: 'tactile', text: 'You have learners with low vision. A tactile number line costs only string and glue — tick tactile materials so the plan includes it?', label: 'Add tactile materials', apply: { res: [...ctx.res, 'tactile'] } });
    if (ctx.need('visual') && !ctx.count('visual')) adv.push({ id: 'countv', text: 'How many learners have low vision? With a number, the group plan seats each of them in a front group.', label: null });
    if (!ctx.needs.size) adv.push({ id: 'noneeds', text: 'No learner needs are ticked. Most classes have some shy learners and some absentees — tick them so the plan supports them?', label: 'Add shy and absentees', apply: { needs: ['shy', 'absent'] } });
    if (!b.known) adv.push({ id: 'score', text: 'Readiness is unknown. Enter the class average from the last test (or JHS mock) and the planner estimates the AP, P and HP numbers.', label: null });
    if (b.source === 'manual' && Math.abs(((+S.ap || 0) + (+S.p || 0) + (+S.hp || 0)) - 100) > 0) adv.push({ id: 'sum', text: `Your AP, P and HP percentages add up to ${(+S.ap || 0) + (+S.p || 0) + (+S.hp || 0)}%, not 100%. The planner scales them, but check the numbers.`, label: null });
    if (!S.homework) adv.push({ id: 'hw', text: M.handmade ? 'Homework is switched off. The recurring-decimal method (Worked Examples 1.2 and 1.3) needs practice at home.' : 'Homework is switched off. The Theory and Exercise questions of the Learner Material need practice at home.', label: 'Turn on homework', apply: { homework: true } });
    if (ctx.apHeavy && S.detail !== 'detailed') adv.push({ id: 'detail', text: 'Many learners are approaching proficiency. Adding key questions to each activity helps you check understanding as you go.', label: 'Add key questions', apply: { detail: 'detailed' } });
    if (+S.keyMins > Math.max(L1.D, L2.D) - 10 && S.keyMode !== 'homework') adv.push({ id: 'katime', text: `A ${S.keyMins}-minute key assessment leaves little lesson time. Make it a take-home assignment or give it at the start of Week 2.`, label: 'Make it take-home', apply: { keyMode: 'homework' } });

    if (typeof o.className === 'string') pick('className', o.className.slice(0, 40));
    if (o.programme in PROGRAMMES) pick('programme', o.programme);
    if (typeof o.school === 'string') pick('school', o.school.slice(0, 80));
    if (typeof o.teacher === 'string') pick('teacher', o.teacher.slice(0, 60));
    if (['1', '2', '3'].includes(String(o.term))) pick('term', String(o.term));
    for (const k of ['dur1', 'dur2']) { const v = int(o[k], 30, 180); if (v) pick(k, [40, 60, 70, 80, 90, 120].reduce((a, b) => Math.abs(b - v) < Math.abs(a - v) ? b : a)); }
    if (typeof o.weekStart === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(o.weekStart)) pick('weekStart', o.weekStart);
    for (const k of ['day1', 'day2']) { const v = int(o[k], 0, 4); if (v !== undefined) pick(k, String(v)); }
    if (['manual', 'score', 'unknown'].includes(o.bandMode)) pick('bandMode', o.bandMode);
    const ls = int(o.lastScore, 0, 100); if (ls !== undefined) { pick('lastScore', ls); if (!out.bandMode) out.bandMode = 'score'; }
    for (const k of ['ap', 'p', 'hp']) pick(k, int(o[k], 0, 100));
    if (['movable', 'fixed'].includes(o.seating)) pick('seating', o.seating);
    if (o.groupSize === 'auto' || int(o.groupSize, 3, 8)) pick('groupSize', o.groupSize === 'auto' ? 'auto' : String(int(o.groupSize, 3, 8)));
    if (typeof o.lang === 'string') pick('lang', o.lang.slice(0, 30));
    if (Array.isArray(o.needs)) out.needs = o.needs.filter(k => k in NEEDS);
    if (o.needCounts && typeof o.needCounts === 'object') { const c = {}; for (const [k, v] of Object.entries(o.needCounts)) if (k in NEEDS && int(v, 1, 60)) c[k] = int(v, 1, 60); out.needCounts = { ...(S.needCounts || {}), ...c }; if (!out.needs) out.needs = [...new Set([...(S.needs || []), ...Object.keys(c)])]; }

    return out;
  }

/* ---------------------------------------------------------------------------
   Typesetter: lays the planner out on A4 landscape pages.
   Output: pages as display lists (text runs with exact glyphs, rules, boxes,
   images) that the PDF writer and the SVG preview both draw, so the preview
   is exactly what prints.
--------------------------------------------------------------------------- */
const Typeset = (() => {
  const PAGE = { w: 841.89, h: 595.28, ml: 34, mr: 34, mt: 30, mb: 38 };
  const C = {
    pink: '#F2CEED', deep: '#E59EDC', peach: '#F6C5AC', note: '#FDF1DE', adapt: '#E4EEFA', accent: '#7A1F6E',
    ink: '#111111', grey: '#5A5A5A', border: '#222222', inner: '#9A9A9A', blue: '#1D5C96', sub: '#333333',
    bar: '#F7E8F4', frame: '#B8B0BC', rule: '#C99BC2'
  };

  /* ---------------------------------------------------------- fonts and glyphs */
  const maps = {};
  for (const k of Object.keys(FONTS)) maps[k] = new Map(Object.entries(FONTS[k].map).map(([c, v]) => [+c, v]));
  const SPACES = { 0x2009: 167, 0x2005: 250, 0x2003: 1000, 0x2002: 500, 0x200B: 0, 0x202F: 167 };
  const ALT = { 0x2610: 0x25A1, 0x27C2: 0x22A5, 0x2B1C: 0x25A1, 0x25FB: 0x25A1 };   // symbols drawn with a near twin the fonts have
  function glyph(key, cp) {
    if (cp === 0xA0) cp = 0x20;
    if (cp in ALT) cp = ALT[cp];
    let m = maps[key].get(cp);
    if (m) return [key, m[0], m[1]];
    if (cp in SPACES) return [key, -1, SPACES[cp]];
    m = maps.S.get(cp); if (m) return ['S', m[0], m[1]];
    if (maps.S2) { m = maps.S2.get(cp); if (m) return ['S2', m[0], m[1]]; }
    if (key !== 'R') { m = maps.R.get(cp); if (m) return ['R', m[0], m[1]]; }
    m = maps[key].get(63);
    return [key, m[0], m[1]];
  }
  // split a string into runs that share one font (missing glyphs fall back to the symbol font)
  function shape(str, key) {
    const segs = []; let cur = null;
    for (const ch of str) {
      const [f, gid, adv] = glyph(key, ch.codePointAt(0));
      if (gid < 0) { segs.push({ f, gids: [], str: '', adv, space: true }); cur = null; continue; }
      if (!cur || cur.f !== f) { cur = { f, gids: [], str: '', adv: 0 }; segs.push(cur); }
      cur.gids.push(gid); cur.str += ch; cur.adv += adv;
    }
    return segs;
  }
  const textW = (str, key, size) => shape(str, key).reduce((t, s) => t + s.adv, 0) * size / 1000;
  const fontKey = (b, i) => b && i ? 'BI' : b ? 'B' : i ? 'I' : 'R';
  const ASC = 0.891, DESC = 0.216;

  /* ---------------------------------------------------------- maths boxes
     A box has width w, ascent a, descent d (points, a/d positive) and ops drawn
     relative to its baseline origin (x right, y down). */
  const REL = new Set('=<>≤≥≠≈∈∉⊂⊆⊃∪∩⟹⇒→≡∼≅∣'.split(''));
  const BIN = new Set('+−×÷±·∓'.split(''));
  const OPEN = new Set('([{'.split('')), CLOSE = new Set(')]}'.split(''));
  const inkH = ch => /[0-9A-Zbdfhklt(){}[\]|√∫]/.test(ch) ? 0.69 : /[a-z]/.test(ch) ? 0.46 : /[ℕℤℚℝπ]/.test(ch) ? (ch === 'π' ? 0.46 : 0.69) : 0.6;
  function runBox(str, size, plain, bold, prev0) {
    const ops = []; let x = 0, ink = 0;
    const chars = [...str];
    let prev = prev0 || 'start', bars = 0;
    chars.forEach((ch, i) => {
      if (ch === ' ' && !plain) return;
      let key = bold ? 'B' : 'R';
      if (!plain && (/[A-Za-z]/.test(ch) || /[α-ωπθ]/.test(ch))) key = bold ? 'BI' : 'I';
      let cls = 'ord';
      if (!plain) {
        if (ch === '|') { cls = bars % 2 === 0 ? 'open' : 'close'; bars++; }
        else if (REL.has(ch)) cls = 'rel';
        else if (BIN.has(ch)) cls = (prev === 'start' || prev === 'rel' || prev === 'bin' || prev === 'open' || prev === 'punct') ? 'ord' : 'bin';
        else if (ch === ',' || ch === ';') cls = 'punct';
        else if (OPEN.has(ch)) cls = 'open'; else if (CLOSE.has(ch)) cls = 'close';
      }
      const sp = cls === 'rel' ? 0.28 : cls === 'bin' ? 0.22 : 0;
      x += (prev === 'start' ? 0 : sp) * size;
      const segs = shape(ch, key);
      for (const s of segs) {
        if (!s.space) ops.push({ t: 'text', x, y: 0, f: s.f, size, gids: s.gids, str: s.str });
        x += s.adv * size / 1000;
      }
      x += sp * size + (cls === 'punct' ? 0.16 * size : 0);
      if (ch.trim()) ink = Math.max(ink, inkH(ch));
      prev = cls;
    });
    return { w: x, a: ASC * size, d: DESC * size, ink: ink * size, ops, last: prev };
  }
  function hbox(list, size, bold) {
    const ops = []; let x = 0, a = 0, d = 0, ink = 0, prev = 'start';
    for (const n of list) {
      let b;
      if (n.t === 'run') {
        if (!n.s) continue;
        if (n.big) b = bigOp(n.s, size, bold);
        else { b = runBox(n.s, size, n.plain, bold || n.bold, prev === 'fn' || prev === 'op' ? 'ord' : prev); if (n.fn) b.last = 'fn'; }
      }
      else b = node(n, size, bold);
      // a function name (sin, log, ...) is followed by a thin space unless an opening bracket follows
      if (prev === 'fn' && !(n.t === 'delim' || (n.t === 'run' && /^[(\[{|]/.test(n.s)) || (n.t === 'mat' && n.open))) x += 0.17 * size;
      ops.push(...shift(b, x, 0));
      x += b.w; a = Math.max(a, b.a); d = Math.max(d, b.d); ink = Math.max(ink, b.ink || 0);
      prev = b.last || 'ord';
    }
    return { w: x, a: a || ASC * size, d: d || DESC * size, ink, ops, last: prev };
  }
  function bigOp(ch, size, bold) {
    const bs = size * 1.3;
    const g = runBox(ch, bs, true, bold);
    const dy = 0.12 * size;                       // centre the large sign on the maths axis
    return { w: g.w + 0.1 * size, a: g.a * 0.92, d: g.d + dy, ink: 0.8 * bs, ops: shift(g, 0.03 * size, dy), last: 'op' };
  }
  function matBox(n, s, bold) {
    const cols = Math.max(1, ...n.rows.map(r => r.length));
    const cells = n.rows.map(r => Array.from({ length: cols }, (_, i) => hbox(r[i] || [], s, bold)));
    const cw = Array.from({ length: cols }, (_, i) => Math.max(0, ...cells.map(r => r[i].w)));
    const rh = cells.map(r => ({ a: Math.max(0.69 * s, ...r.map(c => c.a)), d: Math.max(0.22 * s, ...r.map(c => c.d)) }));
    const rgap = n.align === 'c' ? 0.18 * s : 0.3 * s, cgap = n.align === 'c' ? 0.9 * s : 0;
    const H = rh.reduce((t, r) => t + r.a + r.d, 0) + rgap * (rh.length - 1);
    const axis = 0.25 * s;
    const top = -axis - H / 2;
    const ops = []; let y = top;
    const W = cw.reduce((t, w) => t + w, 0) + cgap * (cols - 1);
    cells.forEach((r, ri) => {
      y += rh[ri].a; let x = 0;
      r.forEach((c, ci) => {
        let dx = (cw[ci] - c.w) / 2;
        if (n.align === 'rl') dx = ci % 2 === 0 ? cw[ci] - c.w : 0;
        if (n.align === 'l') dx = 0;
        ops.push(...shift(c, x + dx, y));
        x += cw[ci] + cgap;
      });
      y += rh[ri].d + rgap;
    });
    const body = { w: W, a: -top, d: H + top, ink: -top, ops };
    if (!n.open && !n.close) return body;
    return node({ t: 'delim', open: n.open, close: n.close, body: [{ t: 'box', box: body }] }, s, bold);
  }
  const shift = (b, dx, dy) => b.ops.map(o => {
    const c = { ...o, x: (o.x || 0) + dx, y: (o.y || 0) + dy };
    if (o.t === 'line') { c.x2 = o.x2 + dx; c.y2 = o.y2 + dy; }
    if (o.t === 'poly') c.pts = o.pts.map(([px, py]) => [px + dx, py + dy]);
    return c;
  });
  function node(n, size, bold) {
    const s = size;
    switch (n.t) {
      case 'grp': return hbox(n.body, s, bold || n.bold);
      case 'box': return n.box;
      case 'mat': return matBox(n, s, bold);
      case 'over': case 'under': {
        const body = hbox(n.body, s, bold), lab = hbox(n.t === 'over' ? n.top : n.bot, s * 0.7, bold);
        const w = Math.max(body.w, lab.w) + 0.1 * s;
        const ops = [...shift(body, (w - body.w) / 2, 0)];
        if (n.t === 'over') { const y = -(Math.max(body.ink || 0.5 * s, 0.5 * s) + 0.08 * s + lab.d); ops.push(...shift(lab, (w - lab.w) / 2, y)); return { w, a: Math.max(body.a, -y + lab.a), d: body.d, ink: -y + lab.ink, ops }; }
        const y = body.d + 0.06 * s + lab.a; ops.push(...shift(lab, (w - lab.w) / 2, y)); return { w, a: body.a, d: y + lab.d, ink: body.ink, ops };
      }
      case 'frac': {
        const fs = s * 0.78, num = hbox(n.num, fs, bold), den = hbox(n.den, fs, bold);
        const axis = 0.25 * s, gap = 0.11 * s, th = Math.max(0.45, 0.048 * s), pad = 0.08 * s;
        const w = Math.max(num.w, den.w) + 2 * pad;
        const nb = -(axis + th / 2 + gap + num.d * 0.7), db = -axis + th / 2 + gap + den.a * 0.78;
        const ops = [...shift(num, (w - num.w) / 2, nb), ...shift(den, (w - den.w) / 2, db)];
        if (!n.nobar) ops.push({ t: 'line', x: pad * 0.5, y: -axis, x2: w - pad * 0.5, y2: -axis, lw: th });
        return { w: w + 0.06 * s, a: -nb + num.a * 0.82, d: db + den.d, ink: -nb + num.ink, ops };
      }
      case 'sup': case 'sub': case 'subsup': {
        const base = hbox(n.base, s, bold);
        const fnBase = n.base.length === 1 && n.base[0].fn;
        const ss = s * 0.7;
        const sup = n.sup ? hbox(n.sup, ss, bold) : null, sub = n.sub ? hbox(n.sub, ss, bold) : null;
        const ops = [...base.ops];
        let w = base.w, a = base.a, d = base.d;
        const up = Math.max(0.38 * s, (base.ink || 0.69 * s) - 0.3 * s);
        if (sup) { ops.push(...shift(sup, base.w + 0.03 * s, -up)); a = Math.max(a, up + sup.ink); w = Math.max(w, base.w + 0.03 * s + sup.w); }
        if (sub) { const dn = 0.2 * s; ops.push(...shift(sub, base.w + 0.02 * s, dn)); d = Math.max(d, dn + sub.d); w = Math.max(w, base.w + 0.02 * s + sub.w); }
        return { w: w + 0.04 * s, a, d, ink: sup ? up + sup.ink : base.ink, ops, last: fnBase ? 'fn' : undefined };
      }
      case 'rad': {
        const body = hbox(n.body, s, bold);
        const lw = Math.max(0.5, 0.05 * s);
        const top = -(Math.max(body.ink, 0.69 * s) + 0.13 * s), bot = Math.max(body.d * 0.6, 0.06 * s);
        const sw = 0.52 * s, mid = top + 0.6 * (bot - top);
        const ops = [
          { t: 'poly', pts: [[0.02 * s, mid + 0.04 * s], [0.12 * s, mid - 0.04 * s], [0.27 * s, bot], [sw - 0.02 * s, top], [sw + body.w + 0.08 * s, top]], lw },
          ...shift(body, sw + 0.03 * s, 0)
        ];
        return { w: sw + body.w + 0.14 * s, a: Math.max(body.a, -top + lw), d: Math.max(body.d, bot + lw), ink: -top, ops };
      }
      case 'acc': {
        const body = hbox(n.body, s, bold);
        const ops = [...body.ops];
        const top = (body.ink || 0.69 * s) + 0.14 * s;
        const k = n.kind || (n.chr === '̇' ? 'dot' : 'bar'), lw = 0.05 * s, cx = body.w / 2 + 0.03 * s;
        if (k === 'under') { const y = Math.max(body.d * 0.55, 0.12 * s); ops.push({ t: 'line', x: 0, y, x2: body.w, y2: y, lw }); return { w: body.w, a: body.a, d: Math.max(body.d, y + lw), ink: body.ink, ops, last: body.last }; }
        let extra = 0.08 * s;
        if (k === 'dot') ops.push({ t: 'circle', x: cx, y: -top, r: 0.06 * s });
        else if (k === 'ddot') { ops.push({ t: 'circle', x: cx - 0.1 * s, y: -top, r: 0.055 * s }, { t: 'circle', x: cx + 0.1 * s, y: -top, r: 0.055 * s }); }
        else if (k === 'hat') { const hw = Math.min(0.22 * s, body.w / 2); ops.push({ t: 'poly', pts: [[cx - hw, -top + 0.07 * s], [cx, -top - 0.06 * s], [cx + hw, -top + 0.07 * s]], lw: lw * 0.9 }); }
        else if (k === 'vec' || k === 'arrow' || k === 'arrow2') {
          const x1 = k === 'vec' ? cx - 0.2 * s : 0.04 * s, x2 = k === 'vec' ? cx + 0.22 * s : body.w + 0.02 * s, y = -top - 0.02 * s, h = 0.09 * s;
          ops.push({ t: 'line', x: x1, y, x2, y2: y, lw: lw * 0.9 }, { t: 'poly', pts: [[x2 - 1.4 * h, y - h], [x2, y], [x2 - 1.4 * h, y + h]], lw: lw * 0.9 });
          if (k === 'arrow2') ops.push({ t: 'poly', pts: [[x1 + 1.4 * h, y - h], [x1, y], [x1 + 1.4 * h, y + h]], lw: lw * 0.9 });
          extra = 0.14 * s;
        }
        else if (k === 'tilde') { const hw = Math.min(0.2 * s, body.w / 2); ops.push({ t: 'poly', pts: [[cx - hw, -top + 0.03 * s], [cx - hw / 2, -top - 0.05 * s], [cx + hw / 2, -top + 0.04 * s], [cx + hw, -top - 0.04 * s]], lw: lw * 0.9 }); }
        else ops.push({ t: 'line', x: 0.05 * s, y: -top, x2: body.w - 0.02 * s, y2: -top, lw });
        return { w: body.w, a: Math.max(body.a, top + extra), d: body.d, ink: top + 0.06 * s, ops, last: body.last };
      }
      case 'delim': {
        const body = hbox(n.body, s, bold);
        const hgt = body.a + body.d;
        const big = hgt > 1.3 * s;
        const ds = big ? hgt / 1.15 : s;
        const axis = 0.25 * s;
        const by = big ? -axis + (ds * (ASC - DESC)) / 2 - ds * 0.25 : 0;
        const ops = [];
        let x = 0;
        if (n.open) { const g = runBox(n.open, ds, true); ops.push(...shift(g, 0, by)); x += g.w; }
        ops.push(...shift(body, x + 0.02 * s, 0)); x += body.w + 0.04 * s;
        if (n.close) { const g = runBox(n.close, ds, true); ops.push(...shift(g, x, by)); x += g.w; }
        return { w: x, a: Math.max(body.a, big ? ds * ASC - by : body.a), d: Math.max(body.d, big ? ds * DESC + by : body.d), ink: body.ink, ops };
      }
      default: return { w: 0, a: 0, d: 0, ops: [] };
    }
  }
  const mathCache = new Map();
  function mathBox(tex, size, bold) {
    const k = tex + '|' + size + '|' + (bold ? 1 : 0);
    if (mathCache.has(k)) return mathCache.get(k);
    const b = hbox(TeX.parse(tex).tree, size, bold);
    b.w += 0.02 * size;
    mathCache.set(k, b);
    return b;
  }

  /* ---------------------------------------------------------- paragraphs */
  // style: {size, b, i, color, align}
  function atoms(text, st) {
    const out = [];
    for (const r of inline(text)) {
      if (r.t === 'math') { const bx = mathBox(r.tex, st.size, st.b || r.b); out.push({ k: 'box', w: bx.w, a: bx.a, d: bx.d, box: bx }); continue; }
      const key = fontKey(st.b || r.b, st.i);
      for (const tok of r.s.split(/( +)/)) {
        if (!tok) continue;
        if (/^ +$/.test(tok)) { out.push({ k: 'sp', w: textW(' ', key, st.size) * tok.length }); continue; }
        const segs = shape(tok, key);
        out.push({ k: 'w', segs, w: segs.reduce((t, s) => t + s.adv, 0) * st.size / 1000, a: ASC * st.size, d: DESC * st.size, size: st.size, color: r.b && st.boldColor ? st.boldColor : st.color });
      }
    }
    return out;
  }
  function lines(text, width, st) {
    const at = atoms(text, st);
    const L = []; let cur = [], w = 0, pend = 0;
    const flush = () => { if (cur.length) L.push(cur); cur = []; w = 0; pend = 0; };
    for (const a of at) {
      if (a.k === 'sp') { if (cur.length) pend += a.w; continue; }
      if (cur.length && w + pend + a.w > width + 0.01) flush();
      if (cur.length && pend) { cur.push({ k: 'sp', w: pend }); w += pend; }
      pend = 0;
      cur.push(a); w += a.w;
    }
    flush();
    return L.map(l => {
      const lw = l.reduce((t, a) => t + a.w, 0);
      const a = Math.max(ASC * st.size, ...l.map(x => x.a || 0)), d = Math.max(DESC * st.size, ...l.map(x => x.d || 0));
      return { atoms: l, w: lw, a, d, h: a + d + st.size * (st.lead ?? 0.14) };
    });
  }
  function drawLine(line, x, top, width, st) {
    const ops = [];
    let cx = x + (st.align === 'center' ? (width - line.w) / 2 : st.align === 'right' ? width - line.w : 0);
    const base = top + line.a + st.size * (st.lead ?? 0.14) / 2;
    for (const a of line.atoms) {
      if (a.k === 'w') {
        let sx = cx;
        for (const s of a.segs) {
          if (!s.space && s.gids.length) ops.push({ t: 'text', x: sx, y: base, f: s.f, size: a.size, gids: s.gids, str: s.str, color: a.color || st.color || C.ink });
          sx += s.adv * a.size / 1000;
        }
      } else if (a.k === 'box') {
        for (const o of a.box.ops) {
          const c = { ...o, x: o.x + cx, y: o.y + base, color: st.color || C.ink };
          if (o.t === 'line') { c.x2 = o.x2 + cx; c.y2 = o.y2 + base; }
          if (o.t === 'poly') c.pts = o.pts.map(([px, py]) => [px + cx, py + base]);
          ops.push(c);
        }
      }
      cx += a.w;
    }
    return ops;
  }

  /* ---------------------------------------------------------- blocks -> vertical slices
     A slice is an unbreakable strip of height h that knows how to draw itself. */
  const S = (h, draw, extra = {}) => ({ h, draw, ...extra });
  function paraSlices(text, width, st, o = {}) {
    const ls = lines(text, width - (o.indent || 0), st);
    const out = [];
    const before = o.before || 0, after = o.after ?? 2;
    ls.forEach((ln, i) => {
      const extraTop = i === 0 ? before : 0, extraBot = i === ls.length - 1 ? after : 0;
      out.push(S(ln.h + extraTop + extraBot, (x, y) => {
        const ops = [];
        if (i === 0 && o.marker) {
          const mk = o.marker;
          const mw = mk.ordered ? textW(mk.text, 'R', st.size) : 0;
          const mx = x + (o.indent || 0) - (mk.ordered ? mw + 3 : 8.5);
          const base = y + extraTop + ln.a + st.size * 0.07;
          if (mk.ordered) ops.push(...textOps(mk.text, mx, base, 'R', st.size, st.color || C.ink));
          else ops.push({ t: 'circle', x: mx + 1.6, y: base - st.size * 0.3, r: st.size * 0.14, color: st.markerColor || C.ink });
        }
        ops.push(...drawLine(ln, x + (o.indent || 0), y + extraTop, width - (o.indent || 0), st));
        return ops;
      }, { keep: o.keep && i === ls.length - 1, keepAll: o.keep }));
    });
    // no lonely first or last line: short paragraphs stay whole, long ones keep two lines at each end
    if (o.keep || out.length <= 3) out.forEach((s, i) => { if (i < out.length - 1) s.keep = true; });
    else { out[0].keep = true; out[out.length - 2].keep = true; }
    if (o.keep) out[out.length - 1].keep = true;
    return out;
  }
  function textOps(str, x, y, key, size, color) {
    const ops = []; let cx = x;
    for (const s of shape(str, key)) { if (!s.space) ops.push({ t: 'text', x: cx, y, f: s.f, size, gids: s.gids, str: s.str, color }); cx += s.adv * size / 1000; }
    return ops;
  }
  // a block that must stay in one piece (notes, figures, combined float groups)
  function solid(slices, extra = {}) {
    const h = slices.reduce((t, s) => t + s.h, 0);
    return S(h, (x, y) => { const ops = []; let cy = y; for (const s of slices) { ops.push(...s.draw(x, cy)); cy += s.h; } return ops; }, extra);
  }
  function styleOf(b, base) {
    return { size: b.size || base.size || 10, b: !!base.b, i: !!(b.italic || base.i), color: b.color === 'accent' ? C.accent : (base.color || C.ink), align: b.align || base.align || 'left', boldColor: base.boldColor };
  }
  function blockSlices(b, width, base) {
    const st = styleOf(b, base);
    switch (b.k) {
      case 'p': {
        if (b.note || b.adapt) {
          const pad = 3.5, bar = b.adapt ? 2.4 : 0;
          const inner = paraSlices(b.text, width - 2 * pad - bar, st, { after: 0 });
          const h = inner.reduce((t, s) => t + s.h, 0) + 2 * pad;
          return [S(h + 3 + (b.before || 0), (x, y) => {
            const top = y + 1.5 + (b.before || 0);
            const ops = [{ t: 'rect', x, y: top, w: width, h, fill: b.note ? C.note : C.adapt }];
            if (bar) ops.push({ t: 'rect', x, y: top, w: bar, h, fill: C.blue });
            let cy = top + pad; for (const s of inner) { ops.push(...s.draw(x + pad + bar, cy)); cy += s.h; }
            return ops;
          })];
        }
        return paraSlices(b.text, width, st, { before: b.before || 0, after: b.after ?? 2, keep: b.keep });
      }
      case 'h': {
        const hs = { ...st, size: b.lvl === 1 ? 10.8 : 10, b: true, color: b.lvl === 1 ? C.accent : st.color };
        const out = paraSlices(b.text, width, hs, { before: b.before ?? (b.lvl === 1 ? 5 : 3), after: b.rule ? 1 : 2, keep: true });
        if (b.rule) {
          out.forEach(x => { x.keep = true; });
          out.push(S(4, (x, y) => [{ t: 'line', x, y: y + 1, x2: x + width, y2: y + 1, lw: 0.7, color: C.rule }], { keep: true }));
        }
        return out;
      }
      case 'bar': {
        const padX = 5, padY = 3.2, rs = 8.8, before = b.before ?? 5;
        const rw = b.right ? Math.min(width * 0.5, textW(b.right, 'R', rs)) : 0;
        const inner = paraSlices(`**${b.text}**`, width - 2 * padX - 4 - (rw ? rw + 14 : 0), { size: 10.2, color: C.ink }, { after: 0 });
        const h = inner.reduce((t, x) => t + x.h, 0) + 2 * padY;
        return [S(h + before + 2, (x, y) => {
          const top = y + before;
          const ops = [{ t: 'rect', x, y: top, w: width, h, fill: C.bar }, { t: 'rect', x, y: top, w: 2.8, h, fill: C.accent }];
          let cy = top + padY; for (const sl of inner) { ops.push(...sl.draw(x + padX + 4, cy)); cy += sl.h; }
          if (rw) ops.push(...textOps(b.right, x + width - padX - rw, top + padY + 9.9, 'R', rs, C.grey));
          return ops;
        }, { keep: true })];
      }
      case 'ul': case 'ol': {
        const out = [];
        const lead = b.sub ? st.size * 1.35 : 0;
        const ind = lead + (b.k === 'ol' ? st.size * 1.55 : st.size * 1.25);
        b.items.forEach((it, i) => out.push(...paraSlices(it, width, st, { indent: ind, after: 1.2, marker: b.k === 'ol' ? { ordered: true, text: `${i + 1}.` } : (b.sub ? { ordered: true, text: '–' } : { ordered: false }) })));
        if (out.length) out[out.length - 1].h += b.sub ? 0.5 : 1.5;
        return out;
      }
      case 'math': {
        let bx = mathBox(b.tex, st.size, false);
        if (bx.w > width - 4) bx = mathBox(b.tex, Math.max(st.size * 0.62, st.size * (width - 4) / bx.w), false);
        return [S(bx.a + bx.d + 6, (x, y) => shift(bx, x + (width - bx.w) / 2, y + 3 + bx.a))];
      }
      case 'fig': return figSlices(b, width, base);
      case 'table': return tableSlices(b, width, base);
    }
    return [];
  }
  function figSize(b, width) {
    const f = FIGS[b.id]; if (!f) return null;
    const w = Math.min(width, (b.cm || 8) * 28.346 * 0.92);
    return { w, h: w * f.h / f.w };
  }
  // a figure sits in a light frame with its caption underneath, so it is clearly set apart from the text
  function figSlices(b, width, base) {
    let sz = figSize(b, width); if (!sz) return [];
    const pad = 5;
    if (sz.w > width - 2 * pad) sz = { w: width - 2 * pad, h: (width - 2 * pad) * sz.h / sz.w };
    const cw = Math.max(sz.w, Math.min(width - 2 * pad, sz.w * 1.3));
    const cap = paraSlices(b.caption, cw, { size: 8.6, i: true, color: C.sub, align: 'center' }, { after: 0 });
    const capH = cap.reduce((t, s) => t + s.h, 0);
    const fh = pad + sz.h + 3 + capH + pad;
    return [S(fh + 8, (x, y) => {
      const fw = cw + 2 * pad, fx = x + (width - fw) / 2, fy = y + 4;
      const ops = [{ t: 'rect', x: fx, y: fy, w: fw, h: fh, fill: '#FFFFFF' }, { t: 'img', id: b.id, x: x + (width - sz.w) / 2, y: fy + pad, w: sz.w, h: sz.h }];
      let cy = fy + pad + sz.h + 3;
      for (const s of cap) { ops.push(...s.draw(x + (width - cw) / 2, cy)); cy += s.h; }
      ops.push({ t: 'rect', x: fx, y: fy, w: fw, h: fh, stroke: C.frame, lw: 0.7 });
      return ops;
    }, { fig: true })];
  }
  function tableSlices(b, width, base) {
    const padH = 3.6, padV = 2.4;
    const cw = b.cols.map(p => width * p / 100);
    const hs = { size: 9.2, b: true, color: C.accent, align: b.small ? 'center' : 'left' };
    const bs = { size: 9, color: C.ink, align: b.small ? 'center' : 'left' };
    const content = (c, w, st, sub) => {
      const parts = Array.isArray(c) ? c.flatMap(x => blockSlices(x, w, st)) : paraSlices(String(c), w, st, { after: 0 });
      if (sub) parts.push(...paraSlices(sub, w, { size: 8.3, i: true, color: C.sub, align: st.align }, { after: 0 }));
      return parts;
    };
    // a row knows how to split itself between pages when it is too tall
    const mkRow = (cs, head) => {
      const h = Math.max(...cs.map(sl => sl.reduce((t, x) => t + x.h, 0))) + 2 * padV;
      const slice = S(h, (x, y) => {
        const ops = []; let cx = x;
        cs.forEach((slices, i) => {
          if (head) ops.push({ t: 'rect', x: cx, y, w: cw[i], h: slice.h, fill: b.fill ? '#' + b.fill : C.pink });
          let cy = y + padV; for (const sl of slices) { ops.push(...sl.draw(cx + padH, cy)); cy += sl.h; }
          ops.push({ t: 'rect', x: cx, y, w: cw[i], h: slice.h, stroke: C.inner, lw: 0.5 });
          cx += cw[i];
        });
        return ops;
      }, { trow: true });
      if (!head) slice.split = maxH => {
        const first = [], rest = [];
        let any = false;
        cs.forEach(sl => {
          let acc = 2 * padV, k = 0;
          while (k < sl.length && acc + sl[k].h <= maxH) { acc += sl[k].h; k++; }
          while (k > 0 && k < sl.length && sl[k - 1].keep) k--;
          if (k) any = true;
          first.push(sl.slice(0, k)); rest.push(sl.slice(k));
        });
        if (!any || rest.every(r => !r.length)) return null;
        // do not leave a small remnant (a line or two in one column, the other columns empty) on either page
        const hh = list => list.reduce((t, x) => t + x.h, 0);
        const firstH = Math.max(...first.map(hh)), restH = Math.max(...rest.map(hh));
        if (restH < Math.max(36, 0.22 * h) || firstH < Math.max(36, 0.22 * h)) return null;
        if (b.tiers && h < 260) return null;
        const A = mkRow(first, false), Bp = mkRow(rest, false);
        A.h = maxH; A.thead = slice.thead; Bp.thead = slice.thead;
        return [A, Bp];
      };
      return slice;
    };
    const row = (cells, head, subs) => mkRow(cells.map((c, i) => content(c, cw[i] - 2 * padH, head ? hs : bs, subs && subs[i])), head);
    const out = [];
    const head = b.head ? row(b.head, true, b.sub) : null;
    if (head) { head.keep = true; out.push(head); }
    for (const r of b.rows) { const s = row(r, false); s.thead = head; out.push(s); }
    out.push(S(4, () => []));
    return out;
  }
  // lay out a list of blocks; a narrow figure floats right beside the text that follows it
  function layoutBlocks(blocks, width, base = {}) {
    const list = (blocks || []).filter(Boolean);
    const out = [];
    for (let i = 0; i < list.length; i++) {
      const b = list[i];
      if (b.k === 'fig') {
        const sz = figSize(b, width);
        const rest = list.slice(i + 1);
        if (sz && sz.w <= width * 0.46 && rest.length && rest[0].k !== 'fig' && !(rest[0].k === 'h')) {
          // the figure and the text beside it form one group; `scale` shrinks the figure a little
          // when the group is just too tall for the space left on a page (no half-empty pages)
          const i0 = i;
          const group = (scale, upto) => {
            const fw = sz.w * scale, gap = 14, tw = width - fw - gap;
            const figS = figSlices({ ...b }, fw, base)[0];
            const side = []; let acc = 0, j = i0 + 1;
            while (j < list.length && (upto ? j < upto : acc < figS.h - 6)) {
              const nb = list[j];
              if (nb.k === 'fig' || nb.k === 'h' || nb.act || (nb.k === 'table' && tw < width * 0.55)) break;
              const sl = blockSlices(nb, tw, base);
              side.push(...sl); acc += sl.reduce((t, s) => t + s.h, 0); j++;
            }
            // never leave a heading-like line (keep) as the last thing beside the figure
            while (!upto && j < list.length && list[j - 1].keep && !list[j].act && list[j].k !== 'fig' && list[j].k !== 'h') {
              const sl = blockSlices(list[j], tw, base); side.push(...sl); acc += sl.reduce((t, s) => t + s.h, 0); j++;
            }
            const h = Math.max(figS.h, acc);
            const g = S(h, (x, y) => {
              const ops = []; let cy = y;
              for (const s of side) { ops.push(...s.draw(x, cy)); cy += s.h; }
              ops.push(...figS.draw(x + tw + gap, y + Math.max(0, (h - figS.h) / 2 - 2)));
              return ops;
            }, { fig: true });
            g.end = j;
            return g;
          };
          const g = group(1);
          // too tall for the space left: first try a slightly smaller figure beside the same text,
          // then a smaller figure beside less text, the remaining text continuing at full width
          g.split = maxH => {
            if (sz.w > 120) for (const sc of [0.92, 0.84, 0.76]) { const t = group(sc, g.end); if (t.end === g.end && t.h <= maxH) return [t]; }
            for (let up = g.end - 1; up > i0 + 1; up--) {
              if (list[up - 1].keep) continue;            // never end the group on a label
              for (const sc of sz.w > 120 ? [1, 0.88, 0.76] : [1]) {
                const t = group(sc, up);
                if (t.h > maxH || t.end !== up) continue;
                const more = [];
                for (let m = up; m < g.end; m++) more.push(...blockSlices(list[m], width, base));
                return [t, ...more];
              }
            }
            return null;
          };
          out.push(g);
          i = g.end - 1;
          continue;
        }
      }
      out.push(...blockSlices(b, width, base));
    }
    return out;
  }

  /* ---------------------------------------------------------- the GES template table and pages */
  const LAB = 0.2054;
  const META = [0.2054, 0.2306, 0.1188, 0.0834, 0.123, 0.083, 0.0844, 0.0714];
  function rows(d) {
    const W = PAGE.w - PAGE.ml - PAGE.mr;
    const R = [];

    const txt = (t, w, st, o) => paraSlices(t, w - 10, st, { after: 0, ...(o || {}) });
    R.push(full(txt(`**Name of Subject:** ${d.subject}${d.extra.length ? '      ' + d.extra.join('   ·   ') : ''}`, W, { size: 10.5 }), { solid: true }));

  function paginate(d, footer) {
    const top = PAGE.mt, bottom = PAGE.h - PAGE.mb, X0 = PAGE.ml;
    const pages = [];
    let ops = [], y = top;
    const newPage = () => { pages.push(ops); ops = []; y = top; };
    const cellH = c => c.slices.reduce((t, s) => t + s.h, 0) + 2 * c.padV;
    const drawCells = (cells, y0, h) => {
      for (const c of cells) {
        if (c.fill) ops.push({ t: 'rect', x: X0 + c.x, y: y0, w: c.w, h, fill: c.fill });
        let cy = y0 + c.padV;
        for (const s of c.slicesNow) { ops.push(...s.draw(X0 + c.x + c.padH, cy)); cy += s.h; }
      }
      for (const c of cells) ops.push({ t: 'rect', x: X0 + c.x, y: y0, w: c.w, h, stroke: C.border, lw: 0.7 });
    };
    const R = rows(d);
    for (let r = 0; r < R.length; r++) {
      let row = R[r];
      let cells = row.cells.map(c => ({ ...c, rest: c.slices.slice() }));
      let first = true;
      for (let guard = 0; guard < 60; guard++) {
        const need = Math.max(row.minH || 0, ...cells.map(c => c.rest.reduce((t, s) => t + s.h, 0) + 2 * c.padV));
        let avail = bottom - y;
        // keep header rows with the start of the next row
        if (row.keepNext && first) {
          let nextStart = 0, k = r + 1;
          while (k < R.length && R[k].keepNext) { nextStart += Math.max(R[k].minH || 0, ...R[k].cells.map(cellH)); k++; }
          if (k < R.length) nextStart += Math.min(48, Math.max(...R[k].cells.map(cellH)));
          if (need + nextStart > avail && y > top + 1) { newPage(); avail = bottom - y; }        }
        if (need <= avail) { cells.forEach(c => { c.slicesNow = c.rest; }); drawCells(cells, y, need); y += need; break; }
        if ((row.solid || avail < 60) && y > top + 1) { newPage(); continue; }
        // split the row: each cell takes what fits, respecting keep-with-next and figure groups
        let took = false;
        cells.forEach(c => {
          let acc = 2 * c.padV, k = 0;
          while (k < c.rest.length && acc + c.rest[k].h <= avail) { acc += c.rest[k].h; k++; }
          // a tall table row can itself be split between pages
          if (k < c.rest.length && c.rest[k].split && avail - acc > 50 && !(k > 0 && c.rest[k - 1].keep && c.rest[k - 1] === c.rest[k].thead && false)) {
            const parts = c.rest[k].split(avail - acc);
            if (parts) {
              c.rest.splice(k, 1, ...parts); acc += parts[0].h; k++;
              while (k < c.rest.length && acc + c.rest[k].h <= avail) { acc += c.rest[k].h; k++; }
            }
          }
          while (k > 0 && k < c.rest.length && c.rest[k - 1].keep) k--;
          if (k > 1 && c.rest.length - k === 1 && c.rest[k].h < 40) k--;   // never carry a single short line over
          if (k === 0 && c.rest.length && y <= top + 1) k = 1;   // a slice taller than a page: place it anyway
          c.slicesNow = c.rest.slice(0, k); c.rest = c.rest.slice(k);
          if (c.rest.length && c.rest[0].thead && c.rest[0] !== c.rest[0].thead && c.slicesNow.length && c.slicesNow[c.slicesNow.length - 1] !== c.rest[0].thead) c.rest.unshift(c.rest[0].thead);
          if (k) took = true;
        });
        if (!took && y > top + 1) { cells.forEach(c => { c.rest = [...c.slicesNow, ...c.rest]; }); newPage(); continue; }
        drawCells(cells, y, bottom - y);
        newPage(); first = false;
        // label cells (short) are not repeated; their fill continues
        row = { ...row, keepNext: false };
      }
    }
    pages.push(ops);
    // footers
    const N = pages.length;
    pages.forEach((p, i) => {
      const fy = PAGE.h - PAGE.mb + 18;
      p.push(...textOps(footer, PAGE.ml, fy, 'R', 8, C.grey));
      const t = `Page ${i + 1} of ${N}`;
      p.push(...textOps(t, PAGE.w - PAGE.mr - textW(t, 'R', 8), fy, 'R', 8, C.grey));
      p.push({ t: 'line', x: PAGE.ml, y: fy - 10, x2: PAGE.w - PAGE.mr, y2: fy - 10, lw: 0.4, color: '#BBBBBB' });
    });
    return pages;
  }

  function layout(d) {
    const footer = `${d.subject} · Year ${d.year || 1} · Week ${d.week} · Learning Planner${d.className ? ' · ' + d.className : ''}`;
    return { pages: paginate(d, footer), size: [PAGE.w, PAGE.h] };
  }
  return { layout, PAGE, textWidth: textW, shape };
})();

/* ---------------------------------------------------------------------------
   A small PDF writer for the typeset pages: embedded TrueType subsets
   (Unicode, so ℕ ℤ ℚ ℝ, √, ≠, ₵ and Ghanaian letters print correctly),
   vector rules and maths, lossless images, selectable and searchable text.
   Also draws the same pages as SVG for the on-screen preview.
--------------------------------------------------------------------------- */
const PDFOut = (() => {
  const enc = new TextEncoder();
  const b64bytes = b64 => { const s = atob(b64); const u = new Uint8Array(s.length); for (let i = 0; i < s.length; i++) u[i] = s.charCodeAt(i); return u; };
  const hex2 = n => n.toString(16).padStart(4, '0');
  const num = v => (Math.round(v * 100) / 100).toString();
  const rgb = h => { const x = parseInt(h.slice(1), 16); return [(x >> 16 & 255) / 255, (x >> 8 & 255) / 255, (x & 255) / 255].map(v => num(v)).join(' '); };
  const pdfStr = s => /^[\x20-\x7e]*$/.test(s) ? '(' + s.replace(/[\\()]/g, m => '\\' + m) + ')' : '<FEFF' + [...s].map(c => { const cp = c.codePointAt(0); if (cp < 0x10000) return hex2(cp); const v = cp - 0x10000; return hex2(0xD800 + (v >> 10)) + hex2(0xDC00 + (v & 1023)); }).join('') + '>';

  async function deflate(bytes) {
    if (typeof CompressionStream === 'undefined') return null;
    try {
      const cs = new CompressionStream('deflate');
      const out = new Response(new Blob([bytes]).stream().pipeThrough(cs)).arrayBuffer();
      return new Uint8Array(await out);
    } catch (e) { return null; }
  }
  // read a non-interlaced 8-bit RGB or palette PNG: its IDAT stream is already a zlib stream with PNG predictors
  function pngImage(u) {
    const dv = new DataView(u.buffer, u.byteOffset, u.byteLength);
    let p = 8, w = 0, h = 0, bd = 8, ct = 2, il = 0, plte = null; const idat = [];
    while (p < u.length) {
      const n = dv.getUint32(p), type = String.fromCharCode(u[p + 4], u[p + 5], u[p + 6], u[p + 7]);
      const d = u.subarray(p + 8, p + 8 + n);
      if (type === 'IHDR') { w = dv.getUint32(p + 8); h = dv.getUint32(p + 12); bd = u[p + 16]; ct = u[p + 17]; il = u[p + 20]; }
      else if (type === 'PLTE') plte = d;
      else if (type === 'IDAT') idat.push(d);
      else if (type === 'IEND') break;
      p += 12 + n;
    }
    if (il || !(ct === 2 || ct === 3 || ct === 0)) throw new Error('Unsupported PNG for PDF (colour type ' + ct + ')');
    const total = idat.reduce((t, d) => t + d.length, 0); const data = new Uint8Array(total); let o = 0; for (const d of idat) { data.set(d, o); o += d.length; }
    let cs = '/DeviceRGB', colors = 3;
    if (ct === 3) { cs = `[/Indexed /DeviceRGB ${plte.length / 3 - 1} <${Array.from(plte, b => b.toString(16).padStart(2, '0')).join('')}>]`; colors = 1; }
    if (ct === 0) { cs = '/DeviceGray'; colors = 1; }
    return { w, h, bpc: bd, cs, colors, data };
  }
  const bezCircle = (cx, cy, r) => {
    const k = 0.5523 * r;
    return `${num(cx + r)} ${num(cy)} m ${num(cx + r)} ${num(cy + k)} ${num(cx + k)} ${num(cy + r)} ${num(cx)} ${num(cy + r)} c ${num(cx - k)} ${num(cy + r)} ${num(cx - r)} ${num(cy + k)} ${num(cx - r)} ${num(cy)} c ${num(cx - r)} ${num(cy - k)} ${num(cx - k)} ${num(cy - r)} ${num(cx)} ${num(cy - r)} c ${num(cx + k)} ${num(cy - r)} ${num(cx + r)} ${num(cy - k)} ${num(cx + r)} ${num(cy)} c f`;
  };

  async function make(lay, meta) {
    const [PW, PH] = lay.size;
    const chunks = []; let pos = 0; const offs = [];
    const push = u => { chunks.push(u); pos += u.length; };
    const text = s => push(enc.encode(s));
    let nobj = 0; const objs = [];
    const alloc = () => ++nobj;
    const writeObj = (id, body) => { offs[id] = pos; text(`${id} 0 obj\n`); if (typeof body === 'string') text(body); else body(); text('\nendobj\n'); };
    const stream = async (id, dict, bytes, compress = true) => {
      let data = bytes, filter = '';
      if (compress) { const z = await deflate(bytes); if (z && z.length < bytes.length) { data = z; filter = '/Filter /FlateDecode '; } }
      writeObj(id, () => { text(`<< ${dict} ${filter}/Length ${data.length} >>\nstream\n`); push(data); text('\nendstream'); });
    };
    text('%PDF-1.7\n%âãÏÓ\n'.replace(/[\u0080-ÿ]/g, c => c));
    const catalog = alloc(), pagesId = alloc(), info = alloc();
    // which fonts and images are used
    const usedF = new Set(), usedI = new Set(), usedG = {};
    for (const p of lay.pages) for (const o of p) {
      if (o.t === 'text') { usedF.add(o.f); (usedG[o.f] = usedG[o.f] || new Set()); o.gids.forEach(g => usedG[o.f].add(g)); }
      if (o.t === 'img') usedI.add(o.id);
    }
    const fontRes = {}, fontIds = {};
    let fi = 0;
    for (const k of usedF) { fontIds[k] = { type0: alloc(), cid: alloc(), desc: alloc(), file: alloc(), tou: alloc(), name: 'F' + (++fi) }; }
    const imgIds = {}; let ii = 0;
    for (const id of usedI) imgIds[id] = { obj: alloc(), name: 'Im' + (++ii) };
    const pageIds = lay.pages.map(() => ({ page: alloc(), content: alloc() }));
    // fonts
    for (const k of usedF) {
      const F = FONTS[k], ids = fontIds[k];
      const tag = 'PLAN' + String.fromCharCode(65 + Object.keys(fontIds).indexOf(k)) + 'A+';
      const base = '/' + tag + F.name.replace(/[^A-Za-z0-9-]/g, '');
      const maxG = Math.max(...Object.values(F.map).map(v => v[0]));
      const wArr = new Array(maxG + 1).fill(F.notdef);
      for (const v of Object.values(F.map)) wArr[v[0]] = v[1];
      writeObj(ids.type0, `<< /Type /Font /Subtype /Type0 /BaseFont ${base} /Encoding /Identity-H /DescendantFonts [${ids.cid} 0 R] /ToUnicode ${ids.tou} 0 R >>`);
      writeObj(ids.cid, `<< /Type /Font /Subtype /CIDFontType2 /BaseFont ${base} /CIDSystemInfo << /Registry (Adobe) /Ordering (Identity) /Supplement 0 >> /FontDescriptor ${ids.desc} 0 R /CIDToGIDMap /Identity /DW ${F.notdef} /W [0 [${wArr.join(' ')}]] >>`);
      writeObj(ids.desc, `<< /Type /FontDescriptor /FontName ${base} /Flags ${F.flags} /FontBBox [${F.bbox.join(' ')}] /ItalicAngle ${F.italic} /Ascent ${F.asc} /Descent ${F.desc} /CapHeight ${F.cap} /StemV ${k.includes('B') ? 120 : 80} /FontFile2 ${ids.file} 0 R >>`);
      const ttf = b64bytes(F.b64);
      // FontFile2 needs /Length1 = uncompressed length
      const z = await deflate(ttf);
      writeObj(ids.file, () => { const data = z && z.length < ttf.length ? z : ttf; text(`<< /Length1 ${ttf.length} ${data === z ? '/Filter /FlateDecode ' : ''}/Length ${data.length} >>\nstream\n`); push(data); text('\nendstream'); });
      // ToUnicode for the glyphs used
      const back = {}; for (const [cp, v] of Object.entries(F.map)) if (usedG[k].has(v[0]) && !(v[0] in back)) back[v[0]] = +cp;
      const ents = Object.entries(back).map(([g, cp]) => `<${hex2(+g)}> <${cp < 0x10000 ? hex2(cp) : (v => hex2(0xD800 + (v >> 10)) + hex2(0xDC00 + (v & 1023)))(cp - 0x10000)}>`);
      let cm = '/CIDInit /ProcSet findresource begin\n12 dict begin\nbegincmap\n/CIDSystemInfo << /Registry (Adobe) /Ordering (UCS) /Supplement 0 >> def\n/CMapName /Adobe-Identity-UCS def\n/CMapType 2 def\n1 begincodespacerange\n<0000> <FFFF>\nendcodespacerange\n';
      for (let i = 0; i < ents.length; i += 100) { const part = ents.slice(i, i + 100); cm += `${part.length} beginbfchar\n${part.join('\n')}\nendbfchar\n`; }
      cm += 'endcmap\nCMapName currentdict /CMap defineresource pop\nend\nend';
      await stream(ids.tou, '', enc.encode(cm));
      fontRes[k] = ids.name;
    }
    // images: raw RGB compressed at build time, or the PNG's own compressed pixels (PNG predictors)
    for (const id of usedI) {
      const f = FIGS[id];
      if (f.raw) {
        const data = b64bytes(f.raw);
        writeObj(imgIds[id].obj, () => { text(`<< /Type /XObject /Subtype /Image /Width ${f.pw} /Height ${f.ph} /ColorSpace /DeviceRGB /BitsPerComponent 8 /Filter /FlateDecode /Length ${data.length} >>\nstream\n`); push(data); text('\nendstream'); });
        continue;
      }
      const im = pngImage(b64bytes(f.png));
      writeObj(imgIds[id].obj, () => { text(`<< /Type /XObject /Subtype /Image /Width ${im.w} /Height ${im.h} /ColorSpace ${im.cs} /BitsPerComponent ${im.bpc} /Filter /FlateDecode /DecodeParms << /Predictor 15 /Colors ${im.colors} /BitsPerComponent ${im.bpc} /Columns ${im.w} >> /Length ${im.data.length} >>\nstream\n`); push(im.data); text('\nendstream'); });
    }
    // pages
    const fontDict = '<< ' + Object.entries(fontIds).map(([k, v]) => `/${v.name} ${v.type0} 0 R`).join(' ') + ' >>';
    const xobj = Object.keys(imgIds).length ? '/XObject << ' + Object.values(imgIds).map(v => `/${v.name} ${v.obj} 0 R`).join(' ') + ' >>' : '';
    for (let i = 0; i < lay.pages.length; i++) {
      const ops = [];
      let curFill = null, curStroke = null, curLW = null;
      const fill = c => { if (c !== curFill) { ops.push(rgb(c) + ' rg'); curFill = c; } };
      const strokeC = c => { if (c !== curStroke) { ops.push(rgb(c) + ' RG'); curStroke = c; } };
      const lw = w => { if (w !== curLW) { ops.push(num(w) + ' w'); curLW = w; } };
      for (const o of lay.pages[i]) {
        const Y = v => num(PH - v);
        if (o.t === 'rect') {
          if (o.fill) { fill(o.fill); ops.push(`${num(o.x)} ${Y(o.y + o.h)} ${num(o.w)} ${num(o.h)} re f`); }
          if (o.stroke) { strokeC(o.stroke); lw(o.lw || 0.6); ops.push(`${num(o.x)} ${Y(o.y + o.h)} ${num(o.w)} ${num(o.h)} re S`); }
        } else if (o.t === 'line') {
          strokeC(o.color || '#111111'); lw(o.lw || 0.6);
          ops.push(`${num(o.x)} ${Y(o.y)} m ${num(o.x2)} ${Y(o.y2)} l S`);
        } else if (o.t === 'poly') {
          strokeC(o.color || '#111111'); lw(o.lw || 0.6);
          ops.push('1 J 1 j ' + o.pts.map(([x, y], k) => `${num(x)} ${Y(y)} ${k ? 'l' : 'm'}`).join(' ') + ' S 0 J 0 j');
        } else if (o.t === 'circle') {
          fill(o.color || '#111111'); ops.push(bezCircle(o.x, PH - o.y, o.r));
        } else if (o.t === 'text') {
          fill(o.color || '#111111');
          ops.push(`BT /${fontRes[o.f]} ${num(o.size)} Tf 1 0 0 1 ${num(o.x)} ${Y(o.y)} Tm <${o.gids.map(hex2).join('')}> Tj ET`);
        } else if (o.t === 'img') {
          ops.push(`q ${num(o.w)} 0 0 ${num(o.h)} ${num(o.x)} ${Y(o.y + o.h)} cm /${imgIds[o.id].name} Do Q`);
        }
      }
      await stream(pageIds[i].content, '', enc.encode(ops.join('\n')));
      writeObj(pageIds[i].page, `<< /Type /Page /Parent ${pagesId} 0 R /MediaBox [0 0 ${num(PW)} ${num(PH)}] /Resources << /Font ${fontDict} ${xobj} >> /Contents ${pageIds[i].content} 0 R >>`);
    }
    writeObj(pagesId, `<< /Type /Pages /Kids [${pageIds.map(p => p.page + ' 0 R').join(' ')}] /Count ${pageIds.length} >>`);
    writeObj(catalog, `<< /Type /Catalog /Pages ${pagesId} 0 R /ViewerPreferences << /DisplayDocTitle true >> >>`);
    const d = new Date(), pad = n => String(n).padStart(2, '0');
    const date = `D:${d.getFullYear()}${pad(d.getMonth() + 1)}${pad(d.getDate())}${pad(d.getHours())}${pad(d.getMinutes())}${pad(d.getSeconds())}`;
    writeObj(info, `<< /Title ${pdfStr(meta.title || 'Learning Planner')} /Subject ${pdfStr(meta.subject || '')} /Creator (Core Maths Planner) /Producer (Core Maths Planner) /CreationDate (${date}) >>`);
    const xref = pos;
    let x = `xref\n0 ${nobj + 1}\n0000000000 65535 f \n`;
    for (let i = 1; i <= nobj; i++) x += String(offs[i]).padStart(10, '0') + ' 00000 n \n';
    text(x + `trailer\n<< /Size ${nobj + 1} /Root ${catalog} 0 R /Info ${info} 0 R >>\nstartxref\n${xref}\n%%EOF\n`);
    const out = new Uint8Array(pos); let o = 0;
    for (const c of chunks) { out.set(c, o); o += c.length; }
    return out;
  }

  /* ---------------------------------------------------------- SVG pages for the preview */
  let fontsReady = null;
  function loadFonts() {
    if (fontsReady) return fontsReady;
    fontsReady = (async () => {

function appendVerifiedSTP(doc,settings){
  // Full original weekly formative assessment, model solutions and grading remain intact.
  clearDiffTablesAndKeepTasks(doc,settings);
  // The original teaching engine uses a generic 'Student Transcript Portal' heading for
  // its weekly key assessment. It is a potentially recordable class exercise, not a
  // scheduled STP milestone. Keep the tasks and marking rubric but label them accurately.
  for (const block of doc.assessment) {
    if(block.k==='h' && /key assessment for the student transcript portal/i.test(block.text||''))
      block.text='2. Weekly key assessment (worked questions and rubric)';
  }
  const firstKey=doc.assessment.findIndex(b=>b.k==='h'&&/2\. Weekly key assessment/.test(b.text||''));
  const rubric=[
    '**Fully secure:** Correct mathematical method, working and final response; award the full question marks stated in the mark scheme.',
    '**Developing:** Uses an appropriate method but makes an arithmetic, notation or explanation error; award earned method marks and explain the correction.',
    '**Needs support:** Missing key concepts or steps; provide one worked example, guided practice and a new question to check understanding.'
  ];
  doc.assessment.splice(firstKey>=0?firstKey:doc.assessment.length,0,
    B.h('Formative feedback rubric',{before:4,keep:true}),B.ul(rubric,{size:9.2}));
  if(Number(doc.year)!==1)return doc;
  const e=STP_ASSESSMENTS_YEAR1[Number(doc.week)]; if(!e)return doc;
  const exemplar=e.sameAs?STP_ASSESSMENTS_YEAR1[e.sameAs]:e;
  doc.assessment.push(B.h(`3. Scheduled STP component — ${e.name}`,{before:7,color:'accent',keep:true}));
  doc.assessment.push(B.p(`**Purpose and administration:** ${e.intro}`,{size:9.5}));
  doc.assessment.push(B.p('**Assessment questions / learner task**',{before:3,keep:true}));
  doc.assessment.push(B.ol(exemplar.tasks,{size:9.7}));
  if([6,12,18,24].includes(+doc.week)){
    const diagram={18:'stp-right-triangle',24:'stp-trapezium'}[Number(doc.week)];
    if(diagram)doc.assessment.push(B.fig(diagram,Number(doc.week)===18?'Diagram: right-angled triangle used for Question 3.':'Diagram: trapezium used for Question 2.',10.5));
  }
  if(+doc.week===13||+doc.week===21)doc.assessment.push(B.fig('stp-coordinate', 'Project guidance: labelled Cartesian axes and two straight lines; learners must construct their own accurately scaled graphs.',10.5));
  doc.assessment.push(B.h('Model answers and marking scheme',{before:5,keep:true}));
  doc.assessment.push(B.ol(exemplar.answers,{size:9.5}));
  doc.assessment.push(B.h('Rubric and feedback guidance',{before:4,keep:true}));
  doc.assessment.push(B.ul(exemplar.rubric,{size:9.4}));
  doc.assessment.push(B.note(`Assessment basis: ${e.source}. The questions and worked marking scheme are teacher-adapted samples aligned to the weekly Teacher Manual content; verify and moderate before use. STP timing follows the recommended Year 1 PLC schedule.`,{size:8.8}));
  return doc;
}

  function scheduleBuild() {
    clearTimeout(buildTimer);

  async function buildFile(fmt) {
    if (!current || !lay || !WEEKS[E.week]) throw new Error('The selected week is still loading.');
    const k = keyOf();
    if (files.key === k && files[fmt] && !files[fmt].error) return files[fmt];
    const active = files.building[fmt];
    if (active && active.key === k) return active.promise;
    const d = current, L = lay;
    const promise = (async () => {
      let data;
      if (fmt === 'docx') {
        const blob = await DOCX.make(d, TEMPLATE_B64);
        data = await blob.arrayBuffer();
      } else if (fmt === 'pdf') {
        data = await PDFOut.make(L, {
          title: `Learning Planner – ${d.subject} – Year ${d.year || 1}, Week ${d.week}${d.className ? ' – ' + d.className : ''}`,
          subject: 'GES Learning Planner'
        });
      } else throw new Error('Unsupported file format');
      if (!data || !data.byteLength) throw new Error('The generated document is empty');
      if (keyOf() === k) {
        files.key = k;
        files[fmt] = data;
      }
      return data;
    })();
    files.building[fmt] = { key: k, promise };
    try { return await promise; }
    finally {
      if (files.building[fmt] && files.building[fmt].promise === promise) delete files.building[fmt];
      updateStates();
    }
  }

  function guidedPickWeek(id){
    const inf=weekInfo(id);if(!inf||inf.Y.y!==guidedYear)return;
    clearTimeout(timer);
    archiveCurrent();
    restoreWeek(id);
    S.term=semesterForWeek(id);
    S.lastWeek={...(S.lastWeek||{}),[guidedYear]:id};
    guidedWeek=id;
    guidedSync();
    guidedWeekGrid();
    ensureWeek(id).then(()=>{if(guidedWeek===id)guidedCurriculumCheck();}).catch(()=>{if(guidedWeek===id){const n=$('#guidedCurriculumFlag');n.hidden=false;n.textContent='This weekly lesson pack could not be loaded; please choose another week or retry later.';}});
  }
  function guidedFillFields(){
    $$('[data-gk]',gw).forEach(el=>{
      const key=el.dataset.gk;
      const v=key==='detailed'?S.detail==='detailed':S[key];
      if(el.type==='checkbox')el.checked=!!v;
      else el.value=v===undefined||v===null?'':String(v);
    });
    $$('[data-gset]',gw).forEach(el=>el.checked=(S[el.dataset.gset]||[]).includes(el.value));
    guidedShowConditional();
  }
  function guidedShowConditional(){
    $$('[data-gshow]',gw).forEach(el=>el.hidden=el.dataset.gshow!==S.bandMode);
    $('#guidedExtraWrap').hidden=(+S.lessonCount||2)<=2;
    const n=Math.max(0,(+S.boys||0)+(+S.girls||0));
    $('#guidedClassTotal').textContent=n?`${n} learners altogether · ${S.boys||0} boys and ${S.girls||0} girls. Group work will be tailored to this class.`:'Enter the numbers of boys and girls to customise group activities.';
  }
  function guidedSummaries(){
    const inf=guidedWeek?weekInfo(guidedWeek):null;
    const dur=(+S.dur1||60)+(+S.dur2||60)+Math.max(0,(+S.lessonCount||2)-2)*(+S.extraMin||40);
    const setCount=k=>(S[k]||[]).length;
    const txt=[
      ['Year & week',inf?`Year ${inf.Y.y} · Week ${inf.w.w}\n${inf.w.title}`:'Not selected',0],
      ['Teacher & school',`${S.teacher||'Not entered'}\n${S.school||'Not entered'} · ${S.className||'No class'}`,2],
      ['Class information',`${(+S.boys||0)+(+S.girls||0)} learners · ${S.boys||0} boys, ${S.girls||0} girls\nReadiness: ${S.bandMode==='score'?'Last assessment':S.bandMode==='manual'?'Ability groups':'Initial diagnostic'}`,3],
      ['Lesson schedule',`${S.lessonCount||2} sessions · ${dur} minutes\nSemester ${S.term||1}${S.weekStart?' · Week beginning '+S.weekStart:''}`,4],
      ['Teaching preferences',`${setCount('res')} resources · ${setCount('strat')} strategies · ${setCount('needs')} support needs\nAssessment: ${S.keyMode||'class exercise'} (${S.keyMins||20} min)`,5]
    ];
    $('#guidedReview').innerHTML=txt.map(([name,desc,idx])=>`<div class="guided-summary"><div class="guided-summary-head"><strong>${escH(name)}</strong><button type="button" data-gjump="${idx}">Change</button></div><p>${escH(desc)}</p></div>`).join('');
  }
  function guidedError(msg){$('#guidedError').textContent=msg||'';}
  function guidedValidate(){
    if(guidedStep===0&&!guidedYear)return 'Please choose Year 1, 2 or 3.';
    if(guidedStep===1&&!guidedWeek)return 'Please select the week you want to plan.';
    if(guidedStep===2){
      if(!(S.teacher||'').trim())return 'Enter the teacher’s name to continue.';
      if(!(S.school||'').trim())return 'Enter the school name to continue.';
      if(!(S.className||'').trim())return 'Enter the class name to continue.';
    }
    if(guidedStep===3){
      const a=+S.boys,b=+S.girls;
      if(!Number.isInteger(a)||!Number.isInteger(b)||a<0||b<0||a>200||b>200||a+b===0)return 'Enter a valid number of boys and girls (at least one learner in total).';
      if(S.bandMode==='score'&&(S.lastScore===''||+S.lastScore<0||+S.lastScore>100))return 'Enter a valid average between 0 and 100.';
      if(S.bandMode==='manual'&&[S.ap,S.p,S.hp].some(v=>v===''||+v<0||+v>100))return 'Enter the percentages for all three readiness groups.';
      if(S.bandMode==='manual'&&Math.abs((+S.ap||0)+(+S.p||0)+(+S.hp||0)-100)>.5)return 'The three readiness percentages must add up to 100%.';
    }
    return '';
  }
  function guidedShow(step){
    guidedStep=Math.min(6,Math.max(0,step));
    $$('[data-gstep]',gw).forEach(el=>el.hidden=+el.dataset.gstep!==guidedStep);
    $('#guidedProgressName').textContent=`STEP ${guidedStep+1} OF 7 · ${GSTEP[guidedStep].toUpperCase()}`;
    const progress=Math.round(100*(guidedStep+1)/7);
    $('#guidedProgressPct').textContent=`${progress}%`;
    $('#guidedFill').style.width=progress+'%';
    $$('#guidedStepper [data-gtab]').forEach(b=>{
      const i=+b.dataset.gtab;b.disabled=i>guidedStep;
      b.dataset.complete=i<guidedStep?'true':'false';
      b.setAttribute('aria-current',i===guidedStep?'step':'false');
      b.querySelector('.g-dot').textContent=i<guidedStep?'✓':i+1;
    });
    $$('#guidedYears [data-gyear]').forEach(b=>b.setAttribute('aria-pressed',String(+b.dataset.gyear===guidedYear)));
    if(guidedStep===1)guidedWeekGrid();
    if(guidedStep===6)guidedSummaries();
    guidedFillFields();guidedError('');
    $('#guidedBack').disabled=guidedStep===0;
    $('#guidedNext').textContent=guidedStep===6?'✦ Generate lesson plan':'Continue →';
    $('#guidedStepHint').textContent=guidedStep===6?'Review before exporting':`Step ${guidedStep+1} of 7`;
    const pos=gw.getBoundingClientRect().top+window.scrollY-15;
    window.scrollTo({top:Math.max(0,pos),behavior:'instant'});
  }
  $('#guidedWeekSearch').addEventListener('input',guidedWeekGrid);
  gw.addEventListener('click',e=>{
    const yr=e.target.closest('[data-gyear]');
    if(yr){const y=+yr.dataset.gyear;if(guidedYear!==y){guidedYear=y;guidedWeek=null;$('#guidedWeekSearch').value='';}$$('#guidedYears [data-gyear]').forEach(b=>b.setAttribute('aria-pressed',String(+b.dataset.gyear===guidedYear)));guidedError('');return;}
    const wk=e.target.closest('[data-gweek]');if(wk){guidedPickWeek(wk.dataset.gweek);guidedError('');return;}
    const tab=e.target.closest('[data-gtab]');if(tab&&!tab.disabled){guidedShow(+tab.dataset.gtab);return;}
    const jump=e.target.closest('[data-gjump]');if(jump){guidedShow(+jump.dataset.gjump);return;}
  });
  function guidedInput(e){
    const el=e.target;
    if(el.dataset.gk){
      const k=el.dataset.gk;
      guidedField(k,el.type==='checkbox'?el.checked:el.value);
      guidedShowConditional();
    }
    if(el.dataset.gset){
      const k=el.dataset.gset;
      S[k]=$$(`[data-gset="${k}"]`,gw).filter(x=>x.checked).map(x=>x.value);
    }
    save();
    guidedError('');
  }
  gw.addEventListener('input',guidedInput);
  gw.addEventListener('change',guidedInput);
  $('#guidedBack').addEventListener('click',()=>guidedShow(guidedStep-1));
  $('#guidedNext').addEventListener('click',async()=>{
    const problem=guidedValidate();if(problem){guidedError(problem);return;}
    if(guidedStep<6){guidedShow(guidedStep+1);return;}
    // The first visible generation happens only after the final confirmation.
    if (!(await window.CoreMathsGate.allow('generate', guidedWeek || S.week))) return;
    guidedSync();
    document.body.classList.remove('guided-setup','guided-power');
    document.body.classList.add('guided-result');
    $('#guidedToggleAdvanced').setAttribute('aria-pressed','false');
    const inf=weekInfo(S.week);
    $('#guidedResultName').textContent=`Year ${inf.Y.y} · Week ${inf.w.w}: ${inf.w.title}`;
    $('#guidedResultSub').textContent=`${S.className} · ${S.teacher} · ${S.lessonCount||2} lessons. Preview below, then download Word or PDF.`;
    render({noRead:true});
    window.scrollTo({top:0,behavior:'instant'});
  });
  function guidedReturn(step){
    document.body.classList.remove('guided-result','guided-power');document.body.classList.add('guided-setup');
    const inf=weekInfo(S.week);if(inf){guidedYear=inf.Y.y;guidedWeek=inf.w.id;}
    guidedShow(step);
  }
  $('#guidedEditAnswers').addEventListener('click',()=>guidedReturn(0));
  $('#guidedAnotherWeek').addEventListener('click',()=>{guidedReturn(1);guidedWeek=null;$('#guidedWeekSearch').value='';guidedWeekGrid();});
  $('#guidedToggleAdvanced').addEventListener('click',e=>{
    document.body.classList.toggle('guided-power');
    e.currentTarget.setAttribute('aria-pressed',String(document.body.classList.contains('guided-power')));
    e.currentTarget.textContent=document.body.classList.contains('guided-power')?'Hide advanced options':'Advanced options';
  });
  // Make exports and preview unavailable until the final Generate action.
  document.body.classList.add('guided-setup');
  window.__guidedPlannerTest={step:()=>guidedStep,year:()=>guidedYear,week:()=>guidedWeek,mode:()=>document.body.classList.contains('guided-result')?'result':'setup',current:()=>current&&current.week};
  guidedShow(0);

})();
